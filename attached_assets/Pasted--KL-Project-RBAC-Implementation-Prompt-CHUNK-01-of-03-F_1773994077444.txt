## STEP 4 — ROLE-CHECK MIDDLEWARE (New File — Additive)

Create `/src/middleware/rbac.js`:

```javascript
// ============================================================
// KL Project — RBAC Middleware
// FILE: /src/middleware/rbac.js
// ACTION: CREATE NEW FILE
// USAGE: Apply selectively to routes — does NOT auto-apply
// ⚠️  Never replace existing auth middleware — ADD alongside it
// ============================================================

const { ROLES, ROLE_LEVEL } = require('../constants/roles');

/**
 * requireRole(...roles)
 * Usage: router.get('/admin/report', authMiddleware, requireRole('ADMIN', 'SUPER_ADMIN'), handler)
 * ⚠️  Always place AFTER your existing authMiddleware, never before
 */
const requireRole = (...allowedRoles) => {
  return (req, res, next) => {
    try {
      // If no user on request (authMiddleware didn't run), pass through
      // to avoid breaking routes that don't use auth
      if (!req.user) {
        return res.status(401).json({ error: 'Authentication required' });
      }

      const userRole = req.user.role;

      // SUPER_ADMIN always passes
      if (userRole === ROLES.SUPER_ADMIN) return next();

      if (allowedRoles.includes(userRole)) return next();

      return res.status(403).json({
        error: 'Access denied',
        required: allowedRoles,
        current: userRole,
      });
    } catch (err) {
      // ⚠️ Never crash existing routes — log and return 403
      console.error('[RBAC] requireRole error:', err.message);
      return res.status(403).json({ error: 'Permission check failed' });
    }
  };
};

/**
 * requireMinLevel(level)
 * Allows any role at or above the specified hierarchy level
 */
const requireMinLevel = (minLevel) => {
  return (req, res, next) => {
    try {
      if (!req.user) return res.status(401).json({ error: 'Authentication required' });
      if (req.user.role === ROLES.SUPER_ADMIN) return next();

      const userLevel = ROLE_LEVEL[req.user.role] || 0;
      if (userLevel >= minLevel) return next();

      return res.status(403).json({ error: 'Insufficient privilege level' });
    } catch (err) {
      console.error('[RBAC] requireMinLevel error:', err.message);
      return res.status(403).json({ error: 'Permission check failed' });
    }
  };
};

/**
 * branchScope
 * Injects branch_id filter into req.query so existing list handlers
 * automatically filter by branch without code changes
 * ⚠️  Apply only to routes that support branch_id filtering
 */
const branchScope = (req, res, next) => {
  try {
    const role = req.user?.role;
    const branchId = req.user?.branch_id;

    // SUPER_ADMIN and ADMIN see all branches — no filter injected
    if (role === ROLES.SUPER_ADMIN || role === ROLES.ADMIN) return next();

    // INVESTOR — filter by investor_branch_scope
    if (role === ROLES.INVESTOR) {
      const scope = req.user?.investor_branch_scope || [];
      if (scope.length > 0) {
        req.rbac_branch_filter = scope; // Array of allowed branch IDs
      }
      return next();
    }

    // All other roles — filter to own branch
    if (branchId) {
      req.query.branch_id = branchId; // Inject into query params
    }
    next();
  } catch (err) {
    console.error('[RBAC] branchScope error:', err.message);
    next(); // Fail open — don't break existing routes
  }
};

/**
 * investorOnly
 * Convenience: blocks all non-investor, non-admin access to financial endpoints
 */
const investorOnly = requireRole(
  ROLES.SUPER_ADMIN,
  ROLES.ADMIN,
  ROLES.INVESTOR
);

module.exports = { requireRole, requireMinLevel, branchScope, investorOnly };
```

// KL Project — Investor Portal API Routes

// ⚠️  These are NEW routes — no existing routes are modified

const express = require('express');
const router  = express.Router();
const { requireRole, branchScope } = require('../middleware/rbac');
const { ROLES } = require('../constants/roles');

// All investor routes require authentication + investor role
// Replace 'authMiddleware' with your actual auth middleware name
const auth = require('../middleware/auth'); // ← YOUR existing auth middleware
const investorAccess = requireRole(ROLES.SUPER_ADMIN, ROLES.ADMIN, ROLES.INVESTOR);

// ── GET /api/investor/reports ─────────────────────────────
// Returns pre-aggregated monthly reports (investor_reports table only)
// NEVER queries live transactional tables
router.get('/reports', auth, investorAccess, branchScope, async (req, res) => {
  try {
    const { period, branch_id } = req.query;
    const userRole = req.user.role;
    const tenantId = req.user.tenant_id;

    // Build safe WHERE clause
    let whereClause = 'WHERE ir.tenant_id = $1';
    const params = [tenantId];
    let paramIndex = 2;

    // Period filter
    if (period) {
      whereClause += ` AND ir.period = $${paramIndex++}`;
      params.push(period);
    }

    // Branch scope for INVESTOR role
    if (userRole === ROLES.INVESTOR) {
      const scope = req.user.investor_branch_scope || [];
      if (scope.length > 0) {
        whereClause += ` AND ir.branch_id = ANY($${paramIndex++})`;
        params.push(scope);
      }
    } else if (branch_id) {
      whereClause += ` AND ir.branch_id = $${paramIndex++}`;
      params.push(branch_id);
    }

    // ⚠️ SELECT only investor-safe fields — never expose raw cost details
    const query = `
      SELECT
        ir.id,
        ir.period,
        ir.branch_name,
        ir.total_revenue,
        ir.room_revenue,
        ir.beverage_revenue,
        ir.food_revenue,
        ir.package_revenue,
        ir.other_revenue,
        ir.total_commission_expense,
        ir.gross_profit,
        ir.net_profit,
        ir.room_utilization_pct,
        ir.total_sessions,
        ir.unique_customers,
        ir.avg_spend_per_session,
        ir.notes,
        ir.currency_code,
        ir.generated_at
        -- ❌ cost_breakdown_detail NOT selected
        -- ❌ staff_salary_total NOT selected
        -- ❌ hostess_names NOT selected
      FROM investor_reports ir
      ${whereClause}
      ORDER BY ir.period DESC, ir.branch_name ASC
    `;

    // Replace db.query with your actual DB client
    const result = await db.query(query, params);
    res.json({ data: result.rows, count: result.rowCount });

  } catch (err) {
    console.error('[Investor] GET /reports error:', err);
    res.status(500).json({ error: 'Failed to fetch investor reports' });
  }
});

// ── GET /api/investor/reports/export/:period ──────────────
// PDF export — watermarked, logged
router.get('/reports/export/:period', auth, investorAccess, async (req, res) => {
  try {

    const investorId = req.user.id;

      `INSERT INTO investor_export_logs

       VALUES ($1, $2, $3, 'PDF', $4)`,
      [investorId, period, req.ip, watermarkText]
    );

    // TODO: Integrate your PDF generation library here
    // For MVP: return JSON with watermark note; replace with PDF in Phase 2

    });

  } catch (err) {
    console.error('[Investor] Export error:', err);
    res.status(500).json({ error: 'Export failed' });
  }

// ── GET /api/investor/kpis ───────────────────────────────
// Aggregated KPI metrics only — no individual records
router.get('/kpis', auth, investorAccess, branchScope, async (req, res) => {
  try {
    const tenantId = req.user.tenant_id;
    const scope = req.user.investor_branch_scope || [];

    let branchFilter = '';
    const params = [tenantId];

    if (req.user.role === ROLES.INVESTOR && scope.length > 0) {
      branchFilter = 'AND branch_id = ANY($2)';
      params.push(scope);
    }

    const query = `
      SELECT
        period,
        SUM(total_revenue)         AS total_revenue,
        AVG(room_utilization_pct)  AS avg_utilization,
        SUM(total_sessions)        AS total_sessions,
        SUM(unique_customers)      AS unique_customers,
        AVG(avg_spend_per_session) AS avg_spend
      FROM investor_reports
      WHERE tenant_id = $1 ${branchFilter}
      GROUP BY period
      ORDER BY period DESC
      LIMIT 12
    `;

    const result = await db.query(query, params);
    res.json({ data: result.rows });

  } catch (err) {
    console.error('[Investor] GET /kpis error:', err);
    res.status(500).json({ error: 'Failed to fetch KPIs' });
  }
});

                {['Period','Branch','Revenue','Gross Profit','Net Profit','Utilization','Export'].map(h => (
                  <th key={h} className="px-4 py-3 text-left text-xs font-medium text-gray-500 uppercase">{h}</th>
                ))}
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-200">
              {reports.map(r => (
                <tr key={r.id} className="hover:bg-gray-50">
                  <td className="px-4 py-3 font-medium">{r.period}</td>
                  <td className="px-4 py-3">{r.branch_name || 'All Branches'}</td>
                  <td className="px-4 py-3">${Number(r.total_revenue).toLocaleString()}</td>
                  <td className="px-4 py-3">${Number(r.gross_profit).toLocaleString()}</td>
                  <td className="px-4 py-3">${Number(r.net_profit).toLocaleString()}</td>
                  <td className="px-4 py-3">{r.room_utilization_pct}%</td>
                  <td className="px-4 py-3">
                    <button
                      onClick={() => window.open(`/api/investor/reports/export/${r.period}`, '_blank')}
                      className="text-blue-600 hover:underline text-xs"
                    >
                      PDF
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};

const KpiCard = ({ label, value }) => (
  <div className="bg-white rounded-lg shadow p-4">
    <p className="text-xs text-gray-500 uppercase tracking-wide">{label}</p>
    <p className="text-2xl font-bold text-gray-900 mt-1">{value}</p>
  </div>
);

export default InvestorDashboard;
