Masking logic function (add to utils/invoiceFormatter.js, create file if not exists):
function getMaskedDisplayName(category, orderItem) {
  switch (category.invoice_display_mode) {
    case 'MASKED_CODE':
      return `SVC-${String(orderItem.sequence_no || orderItem.id.slice(0,3)).padStart(3,'0')}`;
    case 'MASKED_SYMBOL':
      return 'XXXX';
    case 'CUSTOM_ALIAS':
      return category.invoice_alias || '■■■■';
    default:
      return orderItem.name;
  }
}
Export this function. Use it in both order creation and invoice generation.
