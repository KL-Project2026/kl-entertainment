```sql
-- 에이전트(에이전시) 프로파일
CREATE TABLE agents (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  branch_id       UUID REFERENCES branches(id),
  agent_code      VARCHAR(20) UNIQUE NOT NULL,   -- e.g. AGT-001
  company_name    VARCHAR(200) NOT NULL,
  contact_person  VARCHAR(100),
  phone           VARCHAR(30),
  email           VARCHAR(150),
  profile_image_url TEXT,                        -- photo or logo URL
  address         TEXT,
  notes           TEXT,
  is_active       BOOLEAN DEFAULT true,
  
  -- Bank Information (결제/정산 은행 정보)
  bank_name       VARCHAR(100),
  bank_account_number VARCHAR(50),
  bank_account_name   VARCHAR(100),
  bank_swift_code VARCHAR(30),
  bank_country    VARCHAR(10) DEFAULT 'MY',
  
  created_at      TIMESTAMPTZ DEFAULT NOW(),
  updated_at      TIMESTAMPTZ DEFAULT NOW()
);

-- 호스티스↔에이전트 연결 + 커미션 비율 계약
CREATE TABLE agent_hostess_contracts (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id        UUID REFERENCES agents(id) ON DELETE CASCADE,
  hostess_id      UUID REFERENCES hostesses(id) ON DELETE CASCADE,
  
  -- Commission Split (커미션 분배율)
  venue_commission_rate   DECIMAL(5,2) NOT NULL,  -- 호스티스가 받는 % (e.g. 60.00)
  agent_commission_rate   DECIMAL(5,2) NOT NULL,  -- 에이전시가 받는 % (e.g. 40.00)
  -- NOTE: venue_rate + agent_rate = 100.00 always
  
  session_fee_split       DECIMAL(5,2) DEFAULT 60.00, -- 세션피 기준 호스티스 몫
  drink_commission_split  DECIMAL(5,2) DEFAULT 50.00, -- 음료 커미션 기준 호스티스 몫
  package_commission_split DECIMAL(5,2) DEFAULT 55.00,
  
  contract_start  DATE NOT NULL,
  contract_end    DATE,                           -- NULL = indefinite
  is_active       BOOLEAN DEFAULT true,
  
  created_at      TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(agent_id, hostess_id, contract_start)
);

-- 에이전트별 정산 인보이스 (매월/격주 정산)
CREATE TABLE agent_invoices (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id        UUID REFERENCES agents(id),
  invoice_number  VARCHAR(50) UNIQUE NOT NULL,
  period_start    DATE NOT NULL,
  period_end      DATE NOT NULL,
  total_sessions  INT DEFAULT 0,
  gross_revenue   DECIMAL(12,2) DEFAULT 0,
  agent_cut_amount DECIMAL(12,2) DEFAULT 0,
  status          VARCHAR(20) DEFAULT 'pending', -- pending / approved / paid / disputed
  paid_at         TIMESTAMPTZ,
  notes           TEXT,
  created_at      TIMESTAMPTZ DEFAULT NOW()
);
```

Required endpoints:

### Agent CRUD
GET    /api/agents                     → list all agents (branch-scoped)
GET    /api/agents/:id                 → single agent detail
POST   /api/agents                     → create agent
PUT    /api/agents/:id                 → update agent (profile + bank info)
DELETE /api/agents/:id                 → soft delete (set is_active=false)

### Agent ↔ Hostess
GET    /api/agents/:id/hostesses       
  → returns all hostesses under this agent
  → include per-hostess: 
     name, photo, status (active/inactive), 
     venue_commission_rate, agent_commission_rate,
     contract_start, contract_end,
     total_sessions_ytd, total_revenue_ytd, 
     agent_cut_ytd, hostess_earnings_ytd

POST   /api/agents/:id/hostesses       → assign hostess to agent with contract terms
PUT    /api/agents/:id/hostesses/:hostessId → update commission split
DELETE /api/agents/:id/hostesses/:hostessId → remove hostess from agent (end contract)

### Account Summary (Accounts tab within Agent detail)
GET    /api/agents/:id/account-summary?from=YYYY-MM-DD&to=YYYY-MM-DD
  → returns:
  {
    agent: { name, code, profile_image_url },
    period: { from, to },
    summary: {
      total_hostesses: 8,
      total_sessions: 142,
      gross_revenue: 48500.00,       -- all hostess revenue in period
      agent_total_cut: 18250.00,     -- sum of agent commission
      hostess_total_earnings: 30250.00
    },
    hostess_breakdown: [             -- PER HOSTESS LINE ITEMS
      {
        hostess_id, name, photo_url,
        sessions_count: 18,
        gross_revenue: 6200.00,
        agent_cut: 2480.00,          -- gross_revenue × agent_commission_rate
        hostess_earnings: 3720.00,   -- gross_revenue × venue_commission_rate
        commission_rate: "60% / 40%" -- display string
      }
    ]
  }

### Full Revenue Drill-Down (for action button)
GET    /api/agents/:id/hostesses/:hostessId/revenue-detail
  ?from=YYYY-MM-DD&to=YYYY-MM-DD
  → returns full session-level breakdown:
  {
    hostess: { name, photo_url },
    sessions: [
      {
        session_date, room_name, 
        session_hours, session_fee,
        drinks_revenue, package_revenue,
        total_revenue, agent_cut, hostess_earnings,
        notes
      }
    ],
    totals: { ... }
  }

### Profile Image Upload
POST   /api/agents/:id/upload-image
  → multipart/form-data with field "image"
  → store in /public/uploads/agents/ 
  → update agents.profile_image_url in DB
  → return { image_url: '/uploads/agents/filename.jpg' }
