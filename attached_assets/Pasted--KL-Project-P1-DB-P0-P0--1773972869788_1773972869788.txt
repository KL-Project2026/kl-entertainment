## SQL 코드

### 순서 1 — reservations 수정

```sql
-- room_id NOT NULL → nullable (미지정 예약 허용)
-- P0에서 room_id가 NOT NULL이면 실행, 이미 nullable이면 SKIP
ALTER TABLE reservations
  ALTER COLUMN room_id DROP NOT NULL;

-- 배정 관련 컬럼 추가
ALTER TABLE reservations
  ADD COLUMN IF NOT EXISTS table_id    INTEGER,
  ADD COLUMN IF NOT EXISTS assigned_at TIMESTAMPTZ,
  ADD COLUMN IF NOT EXISTS assigned_by INTEGER;
```

### 순서 3 — attendance 테이블

```sql
CREATE TABLE IF NOT EXISTS attendance (
  id           SERIAL PRIMARY KEY,
  person_type  VARCHAR(20)  NOT NULL,
  staff_id     INTEGER,
  hostess_id   INTEGER,
  work_date    DATE         NOT NULL DEFAULT CURRENT_DATE,
  clock_in     TIMESTAMPTZ,
  clock_out    TIMESTAMPTZ,
  hours_worked DECIMAL(5,2),
  status       VARCHAR(20)  DEFAULT 'present',
  late_minutes INTEGER      DEFAULT 0,
  notes        TEXT,
  recorded_by  INTEGER,
  created_at   TIMESTAMPTZ  DEFAULT NOW(),
  updated_at   TIMESTAMPTZ  DEFAULT NOW(),
  CONSTRAINT uq_attendance_person_date
    UNIQUE (person_type, staff_id, hostess_id, work_date)
);

-- hours_worked 자동계산 트리거
CREATE OR REPLACE FUNCTION calc_hours_worked()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.clock_out IS NOT NULL AND NEW.clock_in IS NOT NULL THEN
    NEW.hours_worked := ROUND(
      EXTRACT(EPOCH FROM (NEW.clock_out - NEW.clock_in)) / 3600.0, 2
    );
  END IF;
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_calc_hours ON attendance;
CREATE TRIGGER trg_calc_hours
  BEFORE INSERT OR UPDATE ON attendance
  FOR EACH ROW EXECUTE FUNCTION calc_hours_worked();
```

### 순서 5 — availability_blocks 테이블

```sql
CREATE TABLE IF NOT EXISTS availability_blocks (
  id             SERIAL PRIMARY KEY,
  entity_type    VARCHAR(20) NOT NULL,
  entity_id      INTEGER     NOT NULL,
  block_type     VARCHAR(30) NOT NULL,
  reservation_id INTEGER,
  start_dt       TIMESTAMPTZ NOT NULL,
  end_dt         TIMESTAMPTZ NOT NULL,
  created_by     INTEGER,
  notes          TEXT,
  is_active      BOOLEAN     DEFAULT true,
  created_at     TIMESTAMPTZ DEFAULT NOW(),
  CONSTRAINT chk_block_time CHECK (end_dt > start_dt)
);

CREATE INDEX IF NOT EXISTS idx_avail_entity_time
  ON availability_blocks (entity_type, entity_id, start_dt, end_dt)
  WHERE is_active = true
    AND block_type IN ('booked', 'occupied');
```

### 순서 7 — invoices 테이블

```sql
CREATE TABLE IF NOT EXISTS invoices (
  id               SERIAL PRIMARY KEY,
  invoice_no       VARCHAR(30)   UNIQUE NOT NULL,
  reservation_id   INTEGER       NOT NULL,
  guest_id         INTEGER,
  subtotal         DECIMAL(12,2) DEFAULT 0,
  tax_amount       DECIMAL(12,2) DEFAULT 0,
  discount_amount  DECIMAL(12,2) DEFAULT 0,
  total_amount     DECIMAL(12,2) DEFAULT 0,
  currency         VARCHAR(3)    DEFAULT 'THB',
  status           VARCHAR(20)   DEFAULT 'draft',
  notes            TEXT,
  issued_at        TIMESTAMPTZ,
  issued_by        INTEGER,
  due_date         DATE,
  created_at       TIMESTAMPTZ   DEFAULT NOW(),
  updated_at       TIMESTAMPTZ   DEFAULT NOW()
);

CREATE OR REPLACE FUNCTION generate_invoice_no()
RETURNS VARCHAR AS $$
DECLARE
  today_str VARCHAR := TO_CHAR(NOW(), 'YYYYMMDD');
  seq_num   INTEGER;
BEGIN
  SELECT COUNT(*) + 1 INTO seq_num
  FROM invoices WHERE invoice_no LIKE 'INV-' || today_str || '-%';
  RETURN 'INV-' || today_str || '-' || LPAD(seq_num::TEXT, 4, '0');
END;
$$ LANGUAGE plpgsql;
```
