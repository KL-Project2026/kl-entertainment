### STEP 1 — DATABASE MIGRATION
Run this migration ONLY if the columns/tables do not already exist.
Check with IF NOT EXISTS on all statements.
```sql
-- 1A. Add missing columns to existing menu_categories table (if not already present)
ALTER TABLE menu_categories
  ADD COLUMN IF NOT EXISTS parent_id UUID REFERENCES menu_categories(id) ON DELETE SET NULL,
  ADD COLUMN IF NOT EXISTS icon VARCHAR(50) DEFAULT '🍽️',
  ADD COLUMN IF NOT EXISTS sort_order INT DEFAULT 0,
  ADD COLUMN IF NOT EXISTS is_visible BOOLEAN DEFAULT TRUE,
  ADD COLUMN IF NOT EXISTS tax_rate_override NUMERIC(5,4),
    -- NULL = inherit branch default tax rate
    -- e.g. 0.06 = 6% SST override for this category
  ADD COLUMN IF NOT EXISTS commission_default_rate NUMERIC(5,4) DEFAULT 0,
    -- Default hostess drink commission rate for items in this category
    -- e.g. 0.10 = 10% per item sold
  ADD COLUMN IF NOT EXISTS commission_default_flat NUMERIC(10,2) DEFAULT 0,
    -- Flat MYR commission per item sold (alternative to rate)
    -- commission_default_rate takes priority if both set
  ADD COLUMN IF NOT EXISTS notes TEXT,
  ADD COLUMN IF NOT EXISTS updated_by UUID REFERENCES users(id),
  ADD COLUMN IF NOT EXISTS updated_at TIMESTAMPTZ DEFAULT NOW();

-- 1B. Branch-level category visibility override
-- Allows ADMIN to hide a global category at a specific branch
-- without deleting it or affecting other branches
CREATE TABLE IF NOT EXISTS menu_category_branch_overrides (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  menu_category_id UUID NOT NULL REFERENCES menu_categories(id) ON DELETE CASCADE,
  branch_id UUID NOT NULL REFERENCES branches(id) ON DELETE CASCADE,
  is_visible BOOLEAN NOT NULL DEFAULT TRUE,
  sort_order_override INT,            -- NULL = use global sort_order
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  updated_by UUID REFERENCES users(id),
  UNIQUE(menu_category_id, branch_id)
);

-- 1C. Audit log for all menu config changes (admin-level operations)
CREATE TABLE IF NOT EXISTS menu_config_audit_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  action VARCHAR(50) NOT NULL,        -- 'CATEGORY_CREATED', 'CATEGORY_UPDATED', 
                                      -- 'SUBTYPE_DELETED', 'SORT_ORDER_CHANGED',
                                      -- 'TAX_OVERRIDE_SET', 'BRANCH_VISIBILITY_CHANGED'
  entity_type VARCHAR(50) NOT NULL,   -- 'menu_category'
  entity_id UUID NOT NULL,
  branch_id UUID REFERENCES branches(id),
  changed_by UUID NOT NULL REFERENCES users(id),
  old_value JSONB,                    -- snapshot before change
  new_value JSONB,                    -- snapshot after change
  ip_address VARCHAR(45),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 1D. Indexes
CREATE INDEX IF NOT EXISTS idx_menu_cat_parent ON menu_categories(parent_id);
CREATE INDEX IF NOT EXISTS idx_menu_cat_sort ON menu_categories(sort_order);
CREATE INDEX IF NOT EXISTS idx_menu_cat_branch_override ON menu_category_branch_overrides(branch_id);
CREATE INDEX IF NOT EXISTS idx_menu_config_audit_entity ON menu_config_audit_log(entity_id, created_at DESC);
```
