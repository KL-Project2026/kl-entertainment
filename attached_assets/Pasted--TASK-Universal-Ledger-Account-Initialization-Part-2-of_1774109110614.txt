## STEP 1: Hostess Ledger Accounts 초기화
```sql
-- hostess_profiles 기반 (staff_id 통해 branch/org 조회)
INSERT INTO ledger_accounts (
  org_id, branch_id,
  account_type, entity_id, entity_type,
  currency, is_active, settings
)
SELECT
  s.org_id,               -- ✅ staff.org_id (organizations 참조)
  hp.branch_id,
  'hostess'               AS account_type,
  hp.id                   AS entity_id,   -- hostess_profiles.id
  'hostess_profile'       AS entity_type,
  COALESCE(b.currency, o.base_currency, 'MYR') AS currency,
  TRUE,
  jsonb_build_object(
    'payout_frequency', 'biweekly',
    'advance_limit',    500,
    'agent_id',         hp.agency_id  -- agents.id (null이면 direct)
  )
FROM hostess_profiles hp
JOIN staff s       ON s.id  = hp.staff_id
JOIN organizations o ON o.id = s.org_id
LEFT JOIN branches b ON b.id = hp.branch_id
WHERE hp.deleted_at IS NULL
ON CONFLICT (org_id, account_type, entity_id) DO NOTHING;

SELECT COUNT(*) AS hostess_accounts FROM ledger_accounts
WHERE account_type = 'hostess';
```

NOTE: staff 또는 hostess_profiles에 org_id 컬럼이 없으면
STOP하고 실제 컬럼명 보고.

Report: "✅ Step 1: N hostess accounts created"

## STEP 2: Agent Ledger Accounts 초기화
```sql
-- ✅ agencies → agents
INSERT INTO ledger_accounts (
  org_id, branch_id,
  account_type, entity_id, entity_type,
  currency, is_active, settings
)
SELECT
  a.org_id,          -- agents.org_id
  NULL,              -- Agent는 법인 레벨
  'agent'            AS account_type,
  a.id               AS entity_id,
  'agent'            AS entity_type,
  COALESCE(o.base_currency, 'MYR'),
  TRUE,
  jsonb_build_object(
    'invoice_frequency', 'monthly',
    'invoice_day',       1
  )

WHERE a.deleted_at IS NULL
ON CONFLICT (org_id, account_type, entity_id) DO NOTHING;

SELECT COUNT(*) AS agent_accounts FROM ledger_accounts
WHERE account_type = 'agent';
```

## STEP 3: Driver Ledger Accounts 초기화
```sql
-- ✅ role = 'driver' (소문자)
INSERT INTO ledger_accounts (
  org_id, branch_id,
  account_type, entity_id, entity_type,
  currency, is_active, settings
)
SELECT
  s.org_id,
  s.branch_id,
  'driver'          AS account_type,
  s.id              AS entity_id,
  'staff'           AS entity_type,
  COALESCE(o.base_currency, 'MYR'),
  TRUE,
  jsonb_build_object('payout_frequency', 'weekly')
FROM staff s
JOIN organizations o ON o.id = s.org_id
WHERE s.role = 'driver'
  AND s.is_active = TRUE
  AND s.deleted_at IS NULL
ON CONFLICT (org_id, account_type, entity_id) DO NOTHING;

SELECT COUNT(*) AS driver_accounts FROM ledger_accounts
WHERE account_type = 'driver';
```

Report: "✅ Step 3: N driver accounts created"

## STEP 7: 기존 미지급 잔액 확인 (Opening Balance 사정)
```sql
-- 기존 hostess_payouts 미지급 현황
SELECT
  'hostess_payouts'    AS source,
  COUNT(*)             AS unpaid_count,
  SUM(amount)          AS unpaid_total
FROM hostess_payouts
WHERE status NOT IN ('paid','cancelled')

UNION ALL

-- 기존 agent_payouts 미지급 현황
SELECT
  'agent_payouts',
  COUNT(*),
  SUM(amount)
FROM agent_payouts
WHERE status NOT IN ('paid','cancelled')

UNION ALL

-- 기존 agent_invoices 미지급 현황
SELECT
  'agent_invoices',
  COUNT(*),
  SUM(total_amount)
FROM agent_invoices
WHERE status NOT IN ('paid','cancelled');
```
