## 💱 FX RATES SERVICE

```javascript
// server/features/currency/currency.service.js
// MIGRATION: .NET — CurrencyService with IMemoryCache

const FX_CACHE_TTL_MS = 60 * 60 * 1000; // 60 minutes
let lastFetchTime = 0;

export async function refreshFxRates() {
  const now = Date.now();
  if (now - lastFetchTime < FX_CACHE_TTL_MS) return; // Still fresh

  const res = await fetch(
    `https://v6.exchangerate-api.com/v6/${process.env.EXCHANGERATE_API_KEY}/latest/MYR`
  );
  const data = await res.json();
  if (data.result !== 'success') throw new Error('FX_FETCH_FAILED');

  const supported = ['AUD', 'KRW', 'JPY', 'CNY'];
  for (const ccy of supported) {
    const rate = data.conversion_rates[ccy];
    if (!rate) continue;
    await pool.query(`
      INSERT INTO fx_rates (base_ccy, quote_ccy, rate, fetched_at)
      VALUES ('MYR', $1, $2, NOW())
      ON CONFLICT (base_ccy, quote_ccy) DO UPDATE SET rate=$2, fetched_at=NOW()
    `, [ccy, rate]);
  }
  lastFetchTime = now;
}

export async function convertAmount(amountMyr, toCurrency) {
  if (toCurrency === 'MYR') return { amount: amountMyr, rate: 1.0 };
  await refreshFxRates(); // Only fetches if cache expired
  const { rows } = await pool.query(
    'SELECT rate FROM fx_rates WHERE base_ccy=$1 AND quote_ccy=$2',
    ['MYR', toCurrency]
  );
  const rate = parseFloat(rows[0]?.rate) || 1.0;
  return {
    amount: Math.round(amountMyr * rate * 100) / 100,
    rate,
    note: `Reference rate: 1 MYR = ${rate} ${toCurrency}`,
  };
}
```

```
GET  /api/fx-rates               ← { MYR_AUD: 0.32, MYR_KRW: 295, ... }
POST /api/fx-rates/refresh       ← manual refresh (admin only)
```

```javascript
// server/features/chat/whatsapp.service.js
// MIGRATION: .NET — WhatsAppService

const WHATSAPP_API = 'https://graph.facebook.com/v18.0';

export async function sendBookingConfirmation(reservation, lang = 'en') {
  const phone = reservation.customer_phone?.replace(/[^0-9]/g, '');
  if (!phone) return;

  const messages = {
    en: `✅ *Booking Confirmed!*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count} guests\n\nReply CANCEL to cancel.`,
    zh: `✅ *预订已确认！*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count} 位客人`,
    ko: `✅ *예약 확인*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count}명`,
    ms: `✅ *Tempahan Disahkan!*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count} tetamu`,
    ja: `✅ *予約確認*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count}名様`,
    th: `✅ *ยืนยันการจอง*\n\n📋 *${reservation.reservation_no}*\n📅 ${formatDate(reservation.start_time, lang)}\n🏠 ${reservation.room_name}\n👥 ${reservation.guest_count} คน`,
  };

  await fetch(`${WHATSAPP_API}/${process.env.WHATSAPP_PHONE_NUMBER_ID}/messages`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.WHATSAPP_TOKEN}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      messaging_product: 'whatsapp',
      to: `60${phone}`,  // Malaysia country code
      type: 'text',
      text: { body: messages[lang] || messages['en'] },
    }),
  });
}

export async function sendReceiptWhatsApp(receipt, order, lang = 'en') {
  // Send receipt summary via WhatsApp after payment
  const phone = receipt.customer_phone?.replace(/[^0-9]/g, '');
  if (!phone) return;

  const message = `🧾 *Receipt ${receipt.receipt_no}*\n\n` +
    `Total: MYR ${order.total_amount.toFixed(2)}\n` +
    `Payment: ${order.payment_method}\n` +
    `Date: ${new Date(receipt.payment_at).toLocaleString()}\n\n` +
    `Thank you for visiting! 🙏`;

  await fetch(`${WHATSAPP_API}/${process.env.WHATSAPP_PHONE_NUMBER_ID}/messages`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${process.env.WHATSAPP_TOKEN}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      messaging_product: 'whatsapp',
      to: `60${phone}`,
      type: 'text',
      text: { body: message },
    }),
  });
}
```
