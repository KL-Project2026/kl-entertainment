router.get('/active-sessions', authenticateToken, requireRole(['STORE_MANAGER', 'STAFF', 'ADMIN', 'SUPER_ADMIN']), async (req, res) => {
  try {
    const userRole    = req.user.role;
    const userBranch  = req.user.branchId;
    const queryBranch = req.query.branchId; // "all" or a specific branch ID

    // Branch access control
    // STORE_MANAGER / STAFF → always locked to own branch regardless of query param
    // ADMIN / SUPER_ADMIN   → can query "all" or a specific branch
    let branchFilter;
    if (['STORE_MANAGER', 'STAFF'].includes(userRole)) {
      branchFilter = userBranch;           // force own branch
    } else {
      branchFilter = (queryBranch === 'all' || !queryBranch)
        ? null                             // null = all branches
        : queryBranch;
    }

    // Build dynamic WHERE clause
    const params  = [];
    let   whereClause = `r.status IN ('occupied', 'reserved', 'cleaning', 'maintenance')`;

    if (branchFilter) {
      params.push(branchFilter);
      whereClause += ` AND r.branch_id = $${params.length}`;
    }

    // ⚠️ IMPORTANT: Replace all table/column names below with
    // the ACTUAL names found in the existing codebase (pre-flight check step 1-2)
    const activeSessions = await pool.query(`
      SELECT
        b.id           AS branch_id,
        b.name         AS branch_name,
        r.id           AS room_id,
        r.name         AS room_name,
        rt.name        AS room_type_name,
        r.status       AS room_status,
        rs.id          AS session_id,
        rs.start_time  AS session_start_time,
        rs.expected_end_time,
        o.id           AS order_id,
        o.order_number,
        o.status       AS order_status,
        COALESCE((
          SELECT SUM(oi.quantity * oi.unit_price)
          FROM   order_items oi
          WHERE  oi.order_id = o.id
          AND    oi.status  != 'voided'
        ), 0) AS running_total
      FROM   rooms r
      JOIN   branches  b  ON b.id  = r.branch_id
      LEFT JOIN room_types rt ON rt.id = r.room_type_id
      LEFT JOIN room_sessions rs
             ON rs.room_id = r.id
            AND rs.status IN ('active', 'reserved')
            AND DATE(rs.start_time AT TIME ZONE 'UTC'
                     AT TIME ZONE 'Asia/Kuala_Lumpur') = CURRENT_DATE
      LEFT JOIN orders o
             ON o.session_id = rs.id
            AND o.status NOT IN ('cancelled', 'voided')
      WHERE  ${whereClause}
      ORDER BY
        CASE r.status
          WHEN 'occupied'    THEN 1
          WHEN 'reserved'    THEN 2
          WHEN 'cleaning'    THEN 3
          WHEN 'maintenance' THEN 4
        END,
        rs.start_time ASC NULLS LAST
    `, params);

    res.json({
      success      : true,
      data         : activeSessions.rows,
      totalCount   : activeSessions.rows.length,
      branchFilter : branchFilter || 'all',
      fetchedAt    : new Date().toISOString()
    });

  } catch (error) {
    console.error('[POS Active Sessions] Error:', error);
    res.status(500).json({
      success : false,
      message : 'Failed to fetch active sessions',
      error   : error.message
    });
  }
});
