### State Machine (server/features/reservations/reservations.service.js)

```javascript
// MIGRATION: .NET — ReservationService with IReservationStateMachine
import { VALID_TRANSITIONS } from '../../config/constants.js';
import { emitRoomUpdate } from '../rooms/rooms.socket.js';
import { pool } from '../../config/db.js';

export async function transitionStatus(reservationId, newStatus, userId, io, extra = {}) {
  const { rows } = await pool.query(
    'SELECT * FROM reservations WHERE id = $1', [reservationId]
  );
  const reservation = rows[0];
  if (!reservation) throw new Error('RESERVATION_NOT_FOUND');

  const allowed = VALID_TRANSITIONS[reservation.status] || [];
  if (!allowed.includes(newStatus)) {
    throw new Error(`INVALID_TRANSITION:${reservation.status}→${newStatus}`);
  }

  // Build update fields based on transition
  const updates = { status: newStatus, updated_at: new Date() };
  if (newStatus === 'confirmed')    updates.confirmed_at   = new Date();
  if (newStatus === 'checked_in')   updates.checked_in_at  = new Date();
  if (newStatus === 'checked_out')  updates.checked_out_at = new Date();
  if (newStatus === 'cancelled')  { updates.cancelled_at   = new Date(); updates.cancellation_reason = extra.reason; }
  if (newStatus === 'no_show')      updates.no_show_at     = new Date();
  if (newStatus === 'extended') {
    // Add extra_hours to end_time
    updates.end_time = new Date(reservation.end_time.getTime() + extra.extra_hours * 3600000);
    updates.duration_hours = (reservation.duration_hours || 0) + extra.extra_hours;
  }

  await pool.query(
    `UPDATE reservations SET status=$1, updated_at=$2, confirmed_at=$3,
     checked_in_at=$4, checked_out_at=$5, cancelled_at=$6, cancellation_reason=$7,
     no_show_at=$8, end_time=COALESCE($9, end_time), duration_hours=COALESCE($10, duration_hours)
     WHERE id=$11`,
    [updates.status, updates.updated_at, updates.confirmed_at, updates.checked_in_at,
     updates.checked_out_at, updates.cancelled_at, updates.cancellation_reason,
     updates.no_show_at, updates.end_time, updates.duration_hours, reservationId]
  );

  // Update room status
  const roomStatus = newStatus === 'checked_in' || newStatus === 'extended' ? 'occupied' : 'available';
  await pool.query('UPDATE rooms SET status=$1 WHERE id=$2', [roomStatus, reservation.room_id]);

  // Emit to room board via Socket.io
  if (io && reservation.room_id) {
    emitRoomUpdate(io, reservation.branch_id, {
      id: reservation.room_id,
      status: roomStatus,
      reservationNo: newStatus === 'checked_out' ? null : reservation.reservation_no,
      guestName: newStatus === 'checked_out' ? null : reservation.customer_name,
      guestCount: reservation.guest_count,
      checkInTime: updates.checked_in_at,
      expectedCheckOut: updates.end_time,
    });
  }

  // Write audit log
  await pool.query(
    `INSERT INTO audit_log (entity_type,entity_id,action,changed_by,old_values,new_values)
     VALUES ('reservation',$1,'status_change',$2,$3,$4)`,
    [reservationId, userId,
     JSON.stringify({ status: reservation.status }),
     JSON.stringify({ status: newStatus })]
  );

  return { ...reservation, ...updates };
}

// OUTCALL RULE: if is_outcall=true, driver required before confirm
export async function validateOutcallDriverAssigned(reservationId) {
  const { rows } = await pool.query(
    'SELECT COUNT(*) FROM reservation_pickups WHERE reservation_id=$1 AND status != $2',
    [reservationId, 'cancelled']
  );
  if (parseInt(rows[0].count) === 0) {
    throw new Error('DRIVER_REQUIRED_FOR_OUTCALL');
  }
}
```

### Tax Calculation (orders.service.js)

```javascript
// MIGRATION: .NET — OrderService.CalculateTotals()
export function calculateOrderTotals(items, branchTaxConfig) {
  const { sst_rate = 0.06, service_charge = 0.10 } = branchTaxConfig;

  const subtotal = items.reduce((sum, item) => {
    const discounted = item.unit_price * item.quantity * (1 - (item.discount_pct || 0));
    return sum + discounted;
  }, 0);

  // Only apply SST to tax_applicable items
  const taxableSubtotal = items
    .filter(i => i.tax_applicable !== false)
    .reduce((sum, i) => sum + (i.unit_price * i.quantity * (1 - (i.discount_pct || 0))), 0);

  const sstAmount      = Math.round(taxableSubtotal * sst_rate * 100) / 100;
  const serviceCharge  = Math.round(subtotal * service_charge * 100) / 100;
  const totalAmount    = Math.round((subtotal + sstAmount + serviceCharge) * 100) / 100;

  return { subtotal, sstAmount, serviceCharge, totalAmount };
}
```

### PrintButton Component

```jsx
// client/src/components/documents/PrintButton.jsx
// Props: documentType ('invoice'|'receipt'), documentId, defaultMode ('detailed'|'basic')

export function PrintButton({ documentType, documentId, defaultMode = 'detailed' }) {
  const [mode, setMode] = useState(defaultMode);

  const handlePrint = async () => {
    // 1. Fetch HTML version from API
    const res = await fetch(`/api/${documentType === 'invoice'
      ? `orders/${documentId}/invoice`
      : `receipts/${documentId}`}?mode=${mode}&format=html`);
    const html = await res.text();

    // 2. Open print window
    const win = window.open('', '_blank');
    win.document.write(html);
    win.document.close();
    win.focus();
    win.print();

    // 3. Record print event
    if (documentType === 'receipt') {
      await fetch(`/api/receipts/${documentId}/printed`, { method: 'POST' });
    }
  };

  return (
    <div className="flex items-center gap-3">
      {/* Mode toggle */}
      <div className="flex items-center gap-2 text-sm">
        <span className="text-gray-400">Detailed</span>
        <button
          onClick={() => setMode(m => m === 'detailed' ? 'basic' : 'detailed')}
          className={`relative w-10 h-5 rounded-full transition-colors ${
            mode === 'basic' ? 'bg-yellow-500' : 'bg-blue-600'}`}
        >
          <span className={`absolute top-0.5 w-4 h-4 bg-white rounded-full transition-transform ${
            mode === 'basic' ? 'translate-x-5' : 'translate-x-0.5'}`}/>
        </button>
        <span className="text-gray-400">Basic</span>
      </div>
      <button onClick={handlePrint}
        className="flex items-center gap-2 px-4 py-2 bg-gold text-black rounded-lg font-medium">
        🖨️ Print {documentType === 'invoice' ? 'Invoice' : 'Receipt'}
      </button>
    </div>
  );
}
```
