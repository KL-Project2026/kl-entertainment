## STEP 1: /services/ledger/ledgerAudit.js 생성
```javascript
// /services/ledger/ledgerAudit.js
// ✅ org_id 기반
// MIGRATION: C# BackgroundService (Hangfire)

'use strict';

const ledger = require('./ledgerService');

/**
 * 잔액 정합성 감사
 * balance_cache vs 실제 원장 집계 비교
 */
async function auditAllBalances(db, { orgId, autoFix = false }) {
  const results = [];

  const accounts = await db.query(`
    SELECT id, account_type, entity_id, balance_cache
    FROM ledger_accounts
    WHERE org_id = $1 AND is_active = TRUE
  `, [orgId]);

  for (const acct of accounts.rows) {
    const actual = await db.query(`
      SELECT COALESCE(SUM(
        CASE
          WHEN direction = 'CR' THEN  amount
          WHEN direction = 'DR' THEN -amount
        END
      ), 0.00) AS true_balance
      FROM ledger_entries
      WHERE account_id = $1 AND status = 'posted'
    `, [acct.id]);

    const trueBalance  = parseFloat(actual.rows[0].true_balance);
    const cacheBalance = parseFloat(acct.balance_cache);
    const diff         = Math.abs(trueBalance - cacheBalance);

    if (diff > 0.01) {
      console.warn('[AUDIT] Mismatch:', {
        accountId: acct.id, cache: cacheBalance,
        actual: trueBalance, diff,
      });

      const record = {
        accountId: acct.id, accountType: acct.account_type,
        entityId: acct.entity_id,
        cacheBalance, trueBalance, diff, fixed: false,
      };

      if (autoFix) {
        await ledger.recalculateBalance(db, acct.id);
        record.fixed = true;
      }

      results.push(record);
    }
  }

  return {
    totalChecked: accounts.rows.length,
    mismatches:   results.length,
    autoFixed:    results.filter(r => r.fixed).length,
    details:      results,
  };
}

/**
 * 실패 큐 재시도
 */
async function retryFailedEntries(db, { orgId }) {
  try {
    const failed = await db.query(`
      SELECT * FROM failed_ledger_queue
      WHERE org_id      = $1
        AND retry_count < 3
        AND next_retry_at <= NOW()
      ORDER BY created_at ASC
      LIMIT 50
    `, [orgId]);

    let retried = 0;
    for (const item of failed.rows) {
      try {
        const payload = JSON.parse(item.payload);
        await ledger.postEntry(db, payload);
        await db.query(
          'DELETE FROM failed_ledger_queue WHERE id=$1', [item.id]
        );
        retried++;
      } catch (err) {
        await db.query(`
          UPDATE failed_ledger_queue
          SET retry_count   = retry_count + 1,
              last_error    = $1,
              next_retry_at = NOW() + INTERVAL '30 minutes'
          WHERE id = $2
        `, [err.message, item.id]);
      }
    }

    return { success: true, retried, total: failed.rows.length };
  } catch (err) {
    return { success: false, error: err.message };
  }
}

module.exports = { auditAllBalances, retryFailedEntries };
```

## STEP 2: /services/ledger/ledgerScheduler.js 생성
```javascript
// /services/ledger/ledgerScheduler.js
// ✅ org_id, organizations 기반
// MIGRATION: Hangfire RecurringJob

'use strict';

const { auditAllBalances, retryFailedEntries } = require('./ledgerAudit');
const payslipSvc = require('./payslipService');

async function runNightlyBatch(db, { orgId, branchIds = [] }) {
  const log       = [];
  const startTime = Date.now();

  console.log('[BATCH] Start:', new Date().toISOString(), 'org:', orgId);

  // 1. 잔액 정합성 감사 + 자동 수정
  try {
    const audit = await auditAllBalances(db, { orgId, autoFix: true });
    log.push({ step: 'balance_audit', ...audit });
    console.log('[BATCH] Audit:', audit);
  } catch (err) {
    log.push({ step: 'balance_audit', error: err.message });
  }

  // 2. 실패 항목 재시도
  try {
    const retry = await retryFailedEntries(db, { orgId });
    log.push({ step: 'retry_failed', ...retry });
    console.log('[BATCH] Retry:', retry);
  } catch (err) {
    log.push({ step: 'retry_failed', error: err.message });
  }

  // 3. 매월 1일: 전월 payslip 자동 생성
  const today = new Date();
  if (today.getDate() === 1) {
    const last      = new Date(today.getFullYear(), today.getMonth() - 1, 1);
    const pStart    = last.toISOString().split('T')[0];
    const pEnd      = new Date(
      today.getFullYear(), today.getMonth(), 0
    ).toISOString().split('T')[0];

    for (const branchId of branchIds) {
      for (const acType of ['hostess', 'driver', 'staff']) {
        try {
          const r = await payslipSvc.generateBranchPayslips(db, {
            orgId, branchId, accountType: acType,
            periodStart: pStart, periodEnd: pEnd,
            createdBy: 'system',
          });
          log.push({ step: `payslip_${acType}_${branchId}`, ...r });
        } catch (err) {
          log.push({
            step: `payslip_${acType}_${branchId}`,
            error: err.message,
          });
        }
      }
    }
  }

  const duration = Date.now() - startTime;
  console.log(`[BATCH] Done in ${duration}ms`);
  return { success: true, duration, log };
}

module.exports = { runNightlyBatch };
```

// /routes/ledger.js 하단에 ADD (기존 코드 유지)

const { runNightlyBatch }  = require('../services/ledger/ledgerScheduler');
const { auditAllBalances } = require('../services/ledger/ledgerAudit');

// POST /api/ledger/batch/run — SUPER_ADMIN 전용
router.post('/batch/run',
  requireAuth,
  requireRole(['super_admin']),
  async (req, res) => {
    try {
      const { branch_ids = [] } = req.body;
      // 즉시 응답, 백그라운드 실행
      res.json({
        success: true,
        message: 'Batch started',
        startedAt: new Date().toISOString(),
      });
      runNightlyBatch(db, {
        orgId: req.user.org_id, branchIds: branch_ids,
      }).then(r  => console.log('[BATCH] Complete:', r))
        .catch(e => console.error('[BATCH] Failed:', e.message));
    } catch (err) {
      res.status(500).json({ success: false, error: err.message });
    }
  }
);

// POST /api/ledger/audit/balance-check
router.post('/audit/balance-check',
  requireAuth,
  requireRole(['admin','super_admin']),
  async (req, res) => {
    try {
      const { auto_fix = false } = req.body;
      const r = await auditAllBalances(db, {
        orgId: req.user.org_id, autoFix: auto_fix,
      });
      res.json({ success: true, ...r });
    } catch (err) {
      res.status(500).json({ success: false, error: err.message });
    }
  }
);
