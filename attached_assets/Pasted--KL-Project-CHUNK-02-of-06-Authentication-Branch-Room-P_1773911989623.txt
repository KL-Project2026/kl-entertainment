## 🔐 AUTH (server/features/auth/)

```javascript
// server/features/auth/auth.service.js
// MIGRATION: .NET Identity + JWT Bearer Authentication

import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { pool } from '../../config/db.js';

export async function login(email, password) {
  // Try staff table first, then customers, then shareholders
  const { rows } = await pool.query(
    `SELECT id, full_name AS name, email, password_hash, role, branch_id, 'staff' AS user_type
     FROM staff WHERE email = $1 AND is_active = true AND deleted_at IS NULL
     UNION ALL
     SELECT id, full_name AS name, email, password_hash, 'customer' AS role, NULL, 'customer'
     FROM customers WHERE email = $1 AND is_active = true AND deleted_at IS NULL
     UNION ALL
     SELECT id, name, email, password_hash, 'shareholder' AS role, NULL, 'shareholder'
     FROM shareholders WHERE email = $1 AND is_active = true
     LIMIT 1`,
    [email]
  );
  if (!rows.length) throw new Error('INVALID_CREDENTIALS');
  const user = rows[0];
  const valid = await bcrypt.compare(password, user.password_hash);
  if (!valid) throw new Error('INVALID_CREDENTIALS');

  const accessToken = jwt.sign(
    { id: user.id, role: user.role, branchId: user.branch_id, userType: user.user_type },
    process.env.JWT_SECRET,
    { expiresIn: process.env.JWT_EXPIRY }
  );
  const refreshToken = jwt.sign(
    { id: user.id },
    process.env.REFRESH_TOKEN_SECRET,
    { expiresIn: process.env.REFRESH_TOKEN_EXPIRY }
  );
  return { accessToken, refreshToken, user: { id: user.id, name: user.name, role: user.role } };
}
```

```javascript
// server/middleware/auth.js
import jwt from 'jsonwebtoken';

export function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1] ||
                req.cookies?.accessToken;
  if (!token) return res.status(401).json({ error: 'UNAUTHORIZED' });
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch {
    res.status(401).json({ error: 'TOKEN_EXPIRED' });
  }
}
```

```javascript
// server/middleware/rbac.js
import { ROLES } from '../config/constants.js';

// Usage: router.get('/route', authenticate, requireRole('admin','branch_manager'), handler)
export function requireRole(...allowedRoles) {
  return (req, res, next) => {
    if (allowedRoles.includes('*') || allowedRoles.includes(req.user.role)) return next();
    res.status(403).json({ error: 'FORBIDDEN' });
  };
}

// Branch isolation: branch_manager/manager can only access their own branch
export function requireBranchAccess(req, res, next) {
  const superRoles = [ROLES.SUPER_ADMIN, ROLES.ADMIN];
  if (superRoles.includes(req.user.role)) return next();
  const branchId = req.params.branchId || req.query.branch_id || req.body.branch_id;
  if (branchId && branchId !== req.user.branchId) {
    return res.status(403).json({ error: 'BRANCH_ACCESS_DENIED' });
  }
  next();
}
```

```javascript
// MIGRATION: .NET SignalR — RoomBoardHub

export function initRoomSocket(io) {
  io.on('connection', (socket) => {
    socket.on('join_branch', ({ branchId }) => {
      socket.join(`branch:${branchId}`);
    });
    socket.on('leave_branch', ({ branchId }) => {
      socket.leave(`branch:${branchId}`);
    });
  });
}

// Call this whenever room status changes:
export function emitRoomUpdate(io, branchId, roomData) {
  io.to(`branch:${branchId}`).emit('room_board_update', {
    roomId:          roomData.id,
    roomName:        roomData.name,
    status:          roomData.status,       // available|occupied|cleaning|maintenance
    reservationNo:   roomData.reservationNo || null,
    guestName:       roomData.guestName || null,
    guestCount:      roomData.guestCount || null,
    checkInTime:     roomData.checkInTime || null,
    expectedCheckOut: roomData.expectedCheckOut || null,
    updatedAt:       new Date().toISOString(),
  });
}
```
