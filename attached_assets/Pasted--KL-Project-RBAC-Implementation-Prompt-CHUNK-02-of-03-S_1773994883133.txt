## STEP 1 — STORE MANAGER SCOPED MIDDLEWARE

```javascript
// ============================================================
// FILE: /src/middleware/scopeGuards.js  (NEW FILE)
// ACTION: CREATE — no existing file modified
// PURPOSE: Inject branch_id scope into queries for lower roles
// ============================================================

const { ROLES } = require('../constants/roles');

/**
 * enforceBranchScope
 * For STORE_MANAGER and below: injects branch_id from JWT into req.
 * For ADMIN and SUPER_ADMIN: passes through without restriction.
 *
 * USAGE: Apply to any route that should be branch-scoped.
 * ⚠️  Do NOT apply to routes that already have branch filtering —
 *     check existing handlers before applying.
 */
const enforceBranchScope = (req, res, next) => {
  try {
    const role     = req.user?.role;
    const branchId = req.user?.branch_id;

    const bypassRoles = [ROLES.SUPER_ADMIN, ROLES.ADMIN, ROLES.INVESTOR];
    if (bypassRoles.includes(role)) return next();

    if (!branchId) {
      return res.status(403).json({
        error: 'No branch assigned to this account. Contact your administrator.',
      });
    }

    // Inject scope so handlers can use req.scopedBranchId
    req.scopedBranchId = branchId;

    // Also inject into query for convenience
    // ⚠️  Only if the route supports branch_id query param
    if (!req.query.branch_id) {
      req.query.branch_id = branchId;
    }

    next();
  } catch (err) {
    console.error('[ScopeGuard] enforceBranchScope error:', err.message);
    next(); // Fail open — do not block existing requests
  }
};

/**
 * selfOnlyScope
 * For HOSTESS and DRIVER: injects user_id filter.
 * Used for commission, attendance, assignment endpoints.
 */
const selfOnlyScope = (req, res, next) => {
  try {
    const role   = req.user?.role;
    const userId = req.user?.id;

    const selfRoles = [ROLES.HOSTESS, ROLES.DRIVER];
    if (!selfRoles.includes(role)) return next();

    req.selfOnlyUserId = userId;
    next();
  } catch (err) {
    console.error('[ScopeGuard] selfOnlyScope error:', err.message);
    next();
  }
};

module.exports = { enforceBranchScope, selfOnlyScope };
```

```javascript
// FILE: /src/routes/hostess.js  (NEW FILE)

const express = require('express');
const router  = express.Router();
const auth    = require('../middleware/auth');

// Self-only: hostess sees only her own assignments
router.get('/my-assignments', auth, hostessAccess, selfOnlyScope, async (req, res) => {
  try {
    const userId = req.user.id;
    const role   = req.user.role;

    // If STORE_MANAGER or ADMIN, they may pass a hostess_id in query
    const targetId = (role === ROLES.HOSTESS) ? userId : (req.query.hostess_id || userId);

         ha.id, ha.room_id, ha.started_at, ha.ended_at, ha.status,

       FROM hostess_assignments ha
       JOIN rooms r ON r.id = ha.room_id
       WHERE ha.hostess_id = $1
       ORDER BY ha.started_at DESC
       LIMIT 50`,
      [targetId]
    );

    res.json({ data: result.rows });
  } catch (err) {
    console.error('[Hostess] my-assignments error:', err);
    res.status(500).json({ error: 'Failed to fetch assignments' });
  }
});

// ── GET /api/hostess/my-commissions ──────────────────────
// Self-only: hostess sees only her own commission records
// ⚠️  customer_name is NEVER returned in this endpoint
router.get('/my-commissions', auth, hostessAccess, async (req, res) => {
  try {
    const userId = req.user.id;
    const role   = req.user.role;
    const { period } = req.query;

    const targetId = (role === ROLES.HOSTESS) ? userId : (req.query.hostess_id || userId);

    let dateFilter = '';
    const params = [targetId];

    if (period) {
      dateFilter = `AND DATE_TRUNC('month', c.session_date) = $2`;
      params.push(period);
    } else {
      dateFilter = `AND c.session_date >= NOW() - INTERVAL '3 months'`;
    }

    const result = await db.query(
      `SELECT
         c.id,
         c.session_date,
         c.session_duration,
         c.order_amount,
         c.commission_rate,
         c.commission_amount,
         -- Show agent deduction only if hostess is AGENT type
         CASE WHEN h.hostess_type = 'AGENT' THEN c.agent_fee_deduction ELSE NULL END AS agent_fee_deduction,
         c.net_payable,
         c.status,
         -- ⚠️  customer_name intentionally omitted — privacy
         NULL AS customer_name
       FROM commissions c
       JOIN hostesses h ON h.id = c.hostess_id
       WHERE c.hostess_id = $1 ${dateFilter}
       ORDER BY c.session_date DESC`,
      params
    );

    res.json({ data: result.rows });
  } catch (err) {
    console.error('[Hostess] my-commissions error:', err);
    res.status(500).json({ error: 'Failed to fetch commissions' });
  }
});

// ── POST /api/hostess/clock-in ────────────────────────────
// Clock-in with GPS (GPS is supplementary — failure does not block)
router.post('/clock-in', auth, hostessAccess, async (req, res) => {
  try {
    const userId = req.user.id;
    const { latitude, longitude, notes } = req.body;
    const branchId = req.user.branch_id;

    // Check for existing open attendance record (prevent double clock-in)
    const existing = await db.query(
      `SELECT id FROM attendance
       WHERE user_id = $1 AND clock_out IS NULL
       ORDER BY clock_in DESC LIMIT 1`,
      [userId]
    );

    if (existing.rows.length > 0) {
      return res.status(409).json({
        error: 'Already clocked in. Please clock out first.',
        attendance_id: existing.rows[0].id,
      });
    }

    // Insert new attendance record
    // ⚠️  GPS is nullable — if not provided, clock-in still succeeds
    const result = await db.query(
      `INSERT INTO attendance
         (user_id, branch_id, clock_in, gps_lat_in, gps_lng_in, notes, user_role)
       VALUES ($1, $2, NOW(), $3, $4, $5, $6)
       RETURNING id, clock_in`,
      [userId, branchId, latitude || null, longitude || null, notes || null, req.user.role]
    );

    res.json({
      message: 'Clocked in successfully',
      attendance_id: result.rows[0].id,
      clock_in: result.rows[0].clock_in,
      gps_captured: !!(latitude && longitude),
    });
  } catch (err) {
    console.error('[Hostess] clock-in error:', err);
    res.status(500).json({ error: 'Clock-in failed. Please try again.' });
  }
});

// ── POST /api/hostess/clock-out ───────────────────────────
router.post('/clock-out', auth, hostessAccess, async (req, res) => {
  try {
    const userId = req.user.id;
    const { latitude, longitude } = req.body;

    const result = await db.query(
      `UPDATE attendance
       SET clock_out = NOW(), gps_lat_out = $2, gps_lng_out = $3
       WHERE user_id = $1 AND clock_out IS NULL
       RETURNING id, clock_in, clock_out`,
      [userId, latitude || null, longitude || null]
    );

    if (result.rows.length === 0) {
      return res.status(404).json({ error: 'No open clock-in record found.' });
    }

    res.json({
      message: 'Clocked out successfully',
      attendance_id: result.rows[0].id,
      clock_in:  result.rows[0].clock_in,
      clock_out: result.rows[0].clock_out,
    });
  } catch (err) {
    console.error('[Hostess] clock-out error:', err);
    res.status(500).json({ error: 'Clock-out failed. Please try again.' });
  }
});

module.exports = router;
```

```jsx
// ============================================================
// FILE: /src/pages/HostessDashboard.jsx  (NEW FILE)
// ACTION: CREATE — no existing component modified
// ============================================================

import React, { useState, useEffect } from 'react';

const HostessDashboard = () => {
  const [assignments, setAssignments] = useState([]);
  const [commissions, setCommissions] = useState([]);
  const [clockStatus, setClockStatus] = useState(null); // 'IN' | 'OUT' | null
  const [loading, setLoading]         = useState(true);
  const [actionLoading, setActionLoading] = useState(false);
  const [message, setMessage]         = useState(null);

  const token = localStorage.getItem('token');
  const headers = { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' };

  useEffect(() => {
    const loadData = async () => {
      try {
        const [assignRes, commRes] = await Promise.all([
          fetch('/api/hostess/my-assignments', { headers }),
          fetch('/api/hostess/my-commissions', { headers }),
        ]);
        const assignData = await assignRes.json();
        const commData   = await commRes.json();
        setAssignments(assignData.data || []);
        setCommissions(commData.data || []);
      } catch (err) {
        setMessage({ type: 'error', text: 'Failed to load data. Please refresh.' });
      } finally {
        setLoading(false);
      }
    };
    loadData();
  }, []);

  const handleClockIn = async () => {
    setActionLoading(true);
    try {
      // Request GPS (non-blocking — proceed even if denied)
      let lat = null, lng = null;
      try {
        const pos = await new Promise((resolve, reject) =>
          navigator.geolocation.getCurrentPosition(resolve, reject, { timeout: 5000 })
        );
        lat = pos.coords.latitude;
        lng = pos.coords.longitude;
      } catch {
        // GPS not available — continue without it
      }

      const res = await fetch('/api/hostess/clock-in', {
        method: 'POST',
        headers,
        body: JSON.stringify({ latitude: lat, longitude: lng }),
      });
      const data = await res.json();

      if (res.ok) {
        setClockStatus('IN');
        setMessage({ type: 'success', text: `Clocked in at ${new Date(data.clock_in).toLocaleTimeString()}` });
      } else {
        setMessage({ type: 'error', text: data.error || 'Clock-in failed' });
      }
    } finally {
      setActionLoading(false);
    }
  };

  const handleClockOut = async () => {
    setActionLoading(true);
    try {
      let lat = null, lng = null;
      try {
        const pos = await new Promise((resolve, reject) =>
          navigator.geolocation.getCurrentPosition(resolve, reject, { timeout: 5000 })
        );
        lat = pos.coords.latitude; lng = pos.coords.longitude;
      } catch {}

      const res = await fetch('/api/hostess/clock-out', {
        method: 'POST',
        headers,
        body: JSON.stringify({ latitude: lat, longitude: lng }),
      });
      const data = await res.json();

      if (res.ok) {
        setClockStatus('OUT');
        setMessage({ type: 'success', text: `Clocked out at ${new Date(data.clock_out).toLocaleTimeString()}` });
      } else {
        setMessage({ type: 'error', text: data.error || 'Clock-out failed' });
      }
    } finally {
      setActionLoading(false);
    }
  };

  if (loading) return <div className="p-6 text-center">Loading...</div>;

  const totalCommission = commissions.reduce((sum, c) => sum + parseFloat(c.net_payable || 0), 0);

  return (
    <div className="p-4 max-w-md mx-auto">
      {/* Message Banner */}
      {message && (
        <div className={`mb-4 p-3 rounded-lg text-sm ${
          message.type === 'success' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'
        }`}>
          {message.text}
          <button className="ml-2 font-bold" onClick={() => setMessage(null)}>×</button>
        </div>
      )}

      {/* Clock In/Out */}
      <div className="bg-white rounded-xl shadow p-5 mb-4">
        <h2 className="font-semibold text-gray-700 mb-3">Attendance</h2>
        <div className="flex gap-3">
          <button
            onClick={handleClockIn}
            disabled={actionLoading || clockStatus === 'IN'}
            className="flex-1 bg-green-500 disabled:bg-gray-300 text-white py-3 rounded-lg font-medium"
          >
            {actionLoading ? '...' : 'Clock In'}
          </button>
          <button
            onClick={handleClockOut}
            disabled={actionLoading || clockStatus === 'OUT'}
            className="flex-1 bg-red-500 disabled:bg-gray-300 text-white py-3 rounded-lg font-medium"
          >
            {actionLoading ? '...' : 'Clock Out'}
          </button>
        </div>
      </div>

      {/* Commission Summary */}
      <div className="bg-white rounded-xl shadow p-5 mb-4">
        <h2 className="font-semibold text-gray-700 mb-1">My Commission (3 months)</h2>
        <p className="text-3xl font-bold text-green-600">${totalCommission.toLocaleString()}</p>
        <p className="text-xs text-gray-400 mt-1">{commissions.length} sessions</p>
      </div>

      {/* Assignments */}
      <div className="bg-white rounded-xl shadow p-5">
        <h2 className="font-semibold text-gray-700 mb-3">Recent Assignments</h2>
        {assignments.length === 0
          ? <p className="text-gray-400 text-sm">No assignments yet.</p>
          : assignments.slice(0, 10).map(a => (
            <div key={a.id} className="flex justify-between items-center py-2 border-b last:border-0">
              <span className="font-medium text-sm">Room {a.room_number}</span>
              <span className="text-xs text-gray-500">{new Date(a.started_at).toLocaleDateString()}</span>
              <span className={`text-xs px-2 py-1 rounded-full ${
                a.status === 'ACTIVE' ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'
              }`}>{a.status}</span>
            </div>
          ))
        }
      </div>
    </div>
  );
};

export default HostessDashboard;
```
