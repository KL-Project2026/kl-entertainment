## CONTEXT

The existing `staff` table tracks all employees including hostesses (role = 'HOSTESS').
We now need a SEPARATE `hostess_profiles` table that stores sensitive personal details
ONLY for hostesses, linked 1:1 to the staff record. This separation is deliberate:
- Staff table = operational identity (name, role, branch, login)
- Hostess profile table = personal/commercial details (body stats, services, photos)

This profile data is consumed by:
1. Reservation module → guest requests specific hostess (preview card)
2. Room Order module → hostess assignment panel
3. Hostess self-portal → view own profile
4. Manager panel → manage assignments, guest-hostess matching

### Table 1: hostess_profiles
-- 1:1 with staff table (staff.role = 'HOSTESS')

CREATE TABLE hostess_profiles (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  staff_id            UUID NOT NULL UNIQUE REFERENCES staff(id) ON DELETE CASCADE,
  branch_id           UUID NOT NULL REFERENCES branches(id),

  -- Personal Details (PDPA-sensitive: encrypted at rest in production)
  nationality         VARCHAR(100),           -- e.g. "Malaysian", "Thai", "Korean"
  nationality_code    CHAR(2),                -- ISO 3166-1 alpha-2: MY, TH, KR, CN, SG
  date_of_birth       DATE,                   -- derive age dynamically; never store raw age
  height_cm           SMALLINT,               -- in cm (display conversion handled at API layer)
  weight_kg           DECIMAL(5,2),           -- in kg
  body_size           VARCHAR(10),            -- e.g. "S", "M", "L", "XL", or custom "34-26-36"
  bust_cm             SMALLINT,               -- optional; display only if branch config allows
  waist_cm            SMALLINT,
  hip_cm              SMALLINT,

  -- Profile Content
  intro_text          TEXT,                   -- multilingual; max 500 chars
  intro_translations  JSONB DEFAULT '{}',     -- {"zh": "...", "ko": "...", "th": "...", "en": "..."}
  languages_spoken    TEXT[] DEFAULT '{}',    -- ["en", "zh", "ko", "ms", "th"]

  -- Availability & Status
  status              VARCHAR(20) DEFAULT 'active'
                      CHECK (status IN ('active','inactive','suspended','on_leave')),
  available_today     BOOLEAN DEFAULT false,  -- updated by shift check-in / manager toggle
  
  -- Operational
  display_order       SMALLINT DEFAULT 0,     -- for UI ordering in selection panels
  is_featured         BOOLEAN DEFAULT false,  -- highlight on reservation selection UI
  
  -- Agency link (null = direct hire)
  agency_id           UUID REFERENCES agencies(id),
  agency_hostess_code VARCHAR(50),            -- agency's internal reference code
  
  -- Audit
  pdpa_consent_given  BOOLEAN DEFAULT false,
  pdpa_consent_date   TIMESTAMPTZ,
  pdpa_consent_ip     VARCHAR(45),
  created_by          UUID REFERENCES staff(id),
  updated_by          UUID REFERENCES staff(id),
  created_at          TIMESTAMPTZ DEFAULT now(),
  updated_at          TIMESTAMPTZ DEFAULT now(),
  deleted_at          TIMESTAMPTZ             -- soft delete only; NEVER hard delete
);

-- INDEXES
CREATE INDEX idx_hostess_profiles_branch ON hostess_profiles(branch_id);
CREATE INDEX idx_hostess_profiles_status ON hostess_profiles(status);
CREATE INDEX idx_hostess_profiles_nationality ON hostess_profiles(nationality_code);
CREATE INDEX idx_hostess_profiles_available ON hostess_profiles(branch_id, available_today)
  WHERE status = 'active' AND deleted_at IS NULL;


### Table 2: hostess_services  (CHILD TABLE — 1:many)
-- Each hostess can offer multiple services at different price points

CREATE TABLE hostess_services (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  hostess_profile_id  UUID NOT NULL REFERENCES hostess_profiles(id) ON DELETE CASCADE,
  
  service_code        VARCHAR(50) NOT NULL,   -- e.g. "COMPANION", "SINGING", "POURING", "DANCE"
  service_name        VARCHAR(100) NOT NULL,  -- display name (default language)
  service_translations JSONB DEFAULT '{}',   -- {"zh": "陪唱", "ko": "동반", "th": "..."}
  service_description TEXT,
  
  -- Pricing
  price_amount        DECIMAL(10,2) NOT NULL,
  currency_code       CHAR(3) NOT NULL DEFAULT 'MYR',  -- ISO 4217
  price_unit          VARCHAR(20) DEFAULT 'per_hour'
                      CHECK (price_unit IN ('per_hour','per_session','flat_rate','per_bottle')),
  duration_minutes    SMALLINT,               -- null = unlimited / session-based
  
  -- Availability
  is_active           BOOLEAN DEFAULT true,
  min_booking_hours   SMALLINT DEFAULT 1,
  requires_approval   BOOLEAN DEFAULT false,  -- manager must approve this service type
  
  -- Commission split for this service type
  hostess_commission_pct  DECIMAL(5,2),       -- % of price_amount to hostess
  agency_commission_pct   DECIMAL(5,2),       -- % if agency model (null for direct hire)
  
  display_order       SMALLINT DEFAULT 0,
  created_at          TIMESTAMPTZ DEFAULT now(),
  updated_at          TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_hostess_services_profile ON hostess_services(hostess_profile_id);
CREATE INDEX idx_hostess_services_active ON hostess_services(hostess_profile_id, is_active);


### Table 3: hostess_photos  (CHILD TABLE — 1:many, max 7)
-- Separate table for photo management; supports ordering & feature flagging

CREATE TABLE hostess_photos (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  hostess_profile_id  UUID NOT NULL REFERENCES hostess_profiles(id) ON DELETE CASCADE,
  
  storage_key         VARCHAR(500) NOT NULL,  -- S3/Replit storage path key
  original_filename   VARCHAR(255),
  file_size_bytes     INTEGER,
  mime_type           VARCHAR(50),            -- image/jpeg, image/webp, image/png
  width_px            SMALLINT,
  height_px           SMALLINT,
  
  -- Thumbnail variants (generated on upload)
  thumb_key_sm        VARCHAR(500),           -- 200×200 (list view thumbnail)
  thumb_key_md        VARCHAR(500),           -- 400×400 (card view)
  thumb_key_lg        VARCHAR(500),           -- 800×800 (profile modal / full preview)
  
  is_primary          BOOLEAN DEFAULT false,  -- ONLY ONE per hostess; enforced by trigger
  display_order       SMALLINT DEFAULT 0,     -- 0 = first shown
  is_approved         BOOLEAN DEFAULT false,  -- manager must approve before shown to guests
  approved_by         UUID REFERENCES staff(id),
  approved_at         TIMESTAMPTZ,
  
  upload_by           UUID REFERENCES staff(id),
  created_at          TIMESTAMPTZ DEFAULT now(),
  deleted_at          TIMESTAMPTZ             -- soft delete; preserve for audit
);

-- CONSTRAINT: max 7 active photos per hostess
-- Enforced in API layer + DB trigger (see trigger below)

CREATE INDEX idx_hostess_photos_profile ON hostess_photos(hostess_profile_id);
CREATE INDEX idx_hostess_photos_primary ON hostess_photos(hostess_profile_id, is_primary)
  WHERE is_primary = true AND deleted_at IS NULL;

-- Trigger: ensure only one primary photo per hostess
CREATE OR REPLACE FUNCTION enforce_single_primary_photo()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.is_primary = true THEN
    UPDATE hostess_photos
    SET is_primary = false
    WHERE hostess_profile_id = NEW.hostess_profile_id
      AND id != NEW.id
      AND deleted_at IS NULL;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_single_primary_photo
BEFORE INSERT OR UPDATE ON hostess_photos
FOR EACH ROW EXECUTE FUNCTION enforce_single_primary_photo();

   → available_today is a soft toggle (manager can override for "on call" scenarios)
   → True availability check (for booking) uses shift schedule:
     SELECT hp.* FROM hostess_profiles hp
     JOIN shifts s ON s.staff_id = hp.staff_id
     WHERE s.shift_date = $date
       AND s.start_time <= $requested_start
       AND s.end_time >= $requested_end
       AND NOT EXISTS (
         SELECT 1 FROM hostess_assignments ha
         WHERE ha.hostess_profile_id = hp.id
           AND ha.status IN ('confirmed','in_progress')
           AND ha.start_time < $requested_end
           AND ha.end_time > $requested_start
       )

❌ Never store "age" as a column — always derive from date_of_birth
❌ Never activate a profile without PDPA consent recorded
❌ Never allow more than 7 photos per hostess (enforce in API, not just UI)
❌ Never allow less than 3 approved photos for active status
❌ Never show unapproved photos outside MANAGER+ roles
❌ Never use real name in CUSTOMER-facing endpoints — use display_name alias only
❌ Never calculate commission from live service price — always use price_snapshot at assignment time
❌ Never modify existing staff table structure — hostess_profiles is an EXTENSION, not a replacement
