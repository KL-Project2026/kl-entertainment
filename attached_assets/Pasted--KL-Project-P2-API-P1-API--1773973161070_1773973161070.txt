      SELECT id FROM attendance

        AND ($3::integer IS NULL OR hostess_id=$3)
        AND work_date=$4

      INSERT INTO attendance
        (person_type, staff_id, hostess_id, work_date, clock_in, status, recorded_by, notes)
      VALUES ($1,$2,$3,$4,NOW(),'present',$5,$6) RETURNING *
    `, [person_type, staff_id||null, hostess_id||null, today, recorded_by||null, notes||null]);
    res.json({ success: true, attendance: r.rows[0] });
  } catch (err) {
    if (err.code === '23505') return res.status(409).json({ error: '중복 출근 기록입니다.' });
    res.status(500).json({ error: err.message });
  }
});

// GET /api/availability/rooms?start_dt=...&end_dt=...
router.get('/api/availability/rooms', async (req, res) => {
  const { start_dt, end_dt } = req.query;
  try {
    const r = await db.query(`
      SELECT r.*,
        CASE WHEN EXISTS (
          SELECT 1 FROM availability_blocks ab
          WHERE ab.entity_type='room' AND ab.entity_id=r.id
            AND ab.is_active=true

      WHERE r.is_active=true
      ORDER BY r.room_number
    `, [start_dt, end_dt]);

      available: conflicts.rows.length === 0,
      conflicts: conflicts.rows,

    const start = r.checkin_dt || r.checkin_at || r.start_dt;
    const end   = r.checkout_dt || r.checkout_at || r.end_dt;
    // ⚠️ 컬럼명을 P0 결과로 확인 후 위 변수 수정

        AND block_type IN ('booked','occupied','maintenance','closed')
        AND start_dt < $3 AND end_dt > $2
        AND (reservation_id IS NULL OR reservation_id != $4)

      VALUES ('room',$1,'booked',$2,$3,$4,$5) ON CONFLICT DO NOTHING
    `, [room_id, req.params.id, start, end, assigned_by]);
    await client.query('COMMIT');
    res.json({ success: true, message: '룸이 배정되었습니다.' });

router.post('/api/invoices', async (req, res) => {

  try {
    const entries = await db.query(
      `SELECT * FROM folio_entries WHERE reservation_id=$1 AND is_void=false`,
      [reservation_id]
    );
    const subtotal = entries.rows.reduce((s,e) =>
      !['tax','discount'].includes(e.entry_type) ? s+parseFloat(e.amount) : s, 0);
    const tax      = entries.rows.reduce((s,e) =>
      e.entry_type==='tax'      ? s+parseFloat(e.amount) : s, 0);
    const discount = entries.rows.reduce((s,e) =>
      e.entry_type==='discount' ? s+parseFloat(e.amount) : s, 0);
    const total = subtotal + tax - discount;
    const no = await db.query('SELECT generate_invoice_no() AS no');
    const r  = await db.query(`
      INSERT INTO invoices
        (invoice_no, reservation_id, subtotal, tax_amount,
         discount_amount, total_amount, status, issued_at, issued_by)
      VALUES ($1,$2,$3,$4,$5,$6,'issued',NOW(),$7) RETURNING *
    `, [no.rows[0].no, reservation_id, subtotal, tax, discount, total, issued_by||null]);
    res.json({ success: true, invoice: r.rows[0] });
  } catch (err) { res.status(500).json({ error: err.message }); }
});

// ═══════════════════════════════════════
// PAYMENT & ACCOUNT API

// POST /api/payments — 결제 처리 + 영수증 자동 발행
router.post('/api/payments', async (req, res) => {
  const { invoice_id, amount, method, ref_no, received_by, notes } = req.body;
  const client = await db.connect();
  try {
    await client.query('BEGIN');
    const payment = await client.query(`
      INSERT INTO payments (invoice_id, amount, method, ref_no, received_by, notes)
      VALUES ($1,$2,$3,$4,$5,$6) RETURNING *
    `, [invoice_id, amount, method, ref_no||null, received_by||null, notes||null]);
    const totalPaid = await client.query(
      `SELECT COALESCE(SUM(amount),0) AS paid FROM payments
       WHERE invoice_id=$1 AND is_void=false`, [invoice_id]
    );
    const invoice = await client.query('SELECT * FROM invoices WHERE id=$1', [invoice_id]);
    const paid    = parseFloat(totalPaid.rows[0].paid);
    const due     = parseFloat(invoice.rows[0].total_amount);
    const status  = paid >= due ? 'paid' : 'partially_paid';
    await client.query('UPDATE invoices SET status=$1 WHERE id=$2', [status, invoice_id]);
    let receipt = null;
    if (paid >= due) {
      const rNo = await client.query('SELECT generate_receipt_no() AS no');
      receipt = await client.query(`
        INSERT INTO receipts
          (receipt_no, invoice_id, payment_id, guest_id, total_paid, issued_by)
        SELECT $1, i.id, $2, i.guest_id, $3, $4
        FROM invoices i WHERE i.id=$5 RETURNING *
      `, [rNo.rows[0].no, payment.rows[0].id, paid, received_by||null, invoice_id]);
    }
    await client.query('COMMIT');
    res.json({
      success: true,
      payment: payment.rows[0],
      invoice_status: status,
      receipt: receipt?.rows[0] || null,
    });
  } catch (err) {
    await client.query('ROLLBACK');
    res.status(500).json({ error: err.message });
  } finally { client.release(); }
});
