## 🎯 WHAT TO BUILD IN THIS CHUNK

Set up a Node.js + Express + PostgreSQL project for a **multi-branch KTV (Karaoke) business management platform** called "KL Project".

After this chunk is complete:
- Express server runs on port 3000
- All 17 database tables are created
- Health check endpoint responds
- `.env` file is configured

---

## 🏗️ TECH STACK

```
Runtime:     Node.js 20 LTS
Framework:   Express.js 4
Database:    PostgreSQL (Replit built-in)
Query:       pg (node-postgres) — raw SQL, no ORM
Logging:     Winston
Validation:  Joi
Auth:        jsonwebtoken + bcryptjs (install now, implement in Chunk 2)
```

## 🗄️ DATABASE SCHEMA (migrations/001_schema.sql)

Run this entire file on startup if tables don't exist.

```sql
-- Enable UUID generation
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- ============================================================
-- 1. ORGANIZATIONS
-- ============================================================
CREATE TABLE IF NOT EXISTS organizations (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name          VARCHAR(255) NOT NULL,
  slug          VARCHAR(100) UNIQUE NOT NULL,
  base_currency CHAR(3)      NOT NULL DEFAULT 'MYR',
  default_tz    VARCHAR(60)  NOT NULL DEFAULT 'Asia/Kuala_Lumpur',
  default_lang  VARCHAR(10)  NOT NULL DEFAULT 'en',
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at    TIMESTAMPTZ
);

-- ============================================================
-- 2. BRANCHES
-- ============================================================
CREATE TABLE IF NOT EXISTS branches (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id          UUID         NOT NULL REFERENCES organizations(id),
  name            VARCHAR(255) NOT NULL,
  internal_code   VARCHAR(20)  UNIQUE NOT NULL,
  address         TEXT,
  city            VARCHAR(100),
  country         CHAR(2)      DEFAULT 'MY',
  phone           VARCHAR(50),
  email           VARCHAR(255),
  timezone        VARCHAR(60)  NOT NULL DEFAULT 'Asia/Kuala_Lumpur',
  currency        CHAR(3)      NOT NULL DEFAULT 'MYR',
  operating_hours JSONB,
  tax_config      JSONB        NOT NULL DEFAULT '{"sst_rate":0.06,"service_charge":0.10}',
  settings        JSONB        NOT NULL DEFAULT '{
    "default_invoice_mode":"detailed",
    "default_receipt_mode":"detailed",
    "show_hostess_name_on_invoice":true,
    "show_hostess_name_on_receipt":false,
    "thermal_printer_width_mm":80,
    "invoice_footer_text":"",
    "sst_reg_number":""
  }',
  is_active       BOOLEAN      NOT NULL DEFAULT true,
  created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at      TIMESTAMPTZ
);

-- ============================================================
-- 3. ROOMS
-- ============================================================
CREATE TABLE IF NOT EXISTS rooms (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  branch_id     UUID         NOT NULL REFERENCES branches(id),
  name          VARCHAR(100) NOT NULL,
  room_type     VARCHAR(30)  NOT NULL DEFAULT 'private_room',
  -- VALUES: private_room | vip_room | vvip_room | table | open_area
  capacity_min  SMALLINT     NOT NULL DEFAULT 1,
  capacity_max  SMALLINT     NOT NULL,
  hourly_rate   DECIMAL(15,4),
  min_hours     DECIMAL(4,2) NOT NULL DEFAULT 1.0,
  description   TEXT,
  amenities     JSONB,
  floor_level   VARCHAR(20),
  images        JSONB,
  status        VARCHAR(30)  NOT NULL DEFAULT 'available',
  -- VALUES: available | occupied | cleaning | maintenance | blocked
  sort_order    SMALLINT     NOT NULL DEFAULT 0,
  is_active     BOOLEAN      NOT NULL DEFAULT true,
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at    TIMESTAMPTZ
);
CREATE INDEX IF NOT EXISTS idx_rooms_branch ON rooms(branch_id);

-- ============================================================
-- 4. PRODUCT GROUPS
-- ============================================================
CREATE TABLE IF NOT EXISTS product_groups (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id      UUID     NOT NULL REFERENCES organizations(id),
  name        JSONB    NOT NULL,
  sort_order  SMALLINT NOT NULL DEFAULT 0,
  is_active   BOOLEAN  NOT NULL DEFAULT true,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ============================================================
-- 5. PRODUCT TYPES
-- ============================================================
CREATE TABLE IF NOT EXISTS product_types (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  group_id    UUID     NOT NULL REFERENCES product_groups(id),
  name        JSONB    NOT NULL,
  sort_order  SMALLINT NOT NULL DEFAULT 0,
  is_active   BOOLEAN  NOT NULL DEFAULT true,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ============================================================
-- 6. PRODUCTS
-- ============================================================
CREATE TABLE IF NOT EXISTS products (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  type_id        UUID          NOT NULL REFERENCES product_types(id),
  branch_id      UUID          REFERENCES branches(id),
  sku            VARCHAR(100),
  name           JSONB         NOT NULL,
  description    JSONB,
  unit_price     DECIMAL(15,4) NOT NULL,
  unit           VARCHAR(30)   NOT NULL DEFAULT 'pcs',
  tax_applicable BOOLEAN       NOT NULL DEFAULT true,
  images         JSONB,
  sort_order     SMALLINT      NOT NULL DEFAULT 0,
  is_active      BOOLEAN       NOT NULL DEFAULT true,
  created_at     TIMESTAMPTZ   NOT NULL DEFAULT NOW(),
  deleted_at     TIMESTAMPTZ
);

-- ============================================================
-- 7. AGENTS
-- ============================================================
CREATE TABLE IF NOT EXISTS agents (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id                UUID         NOT NULL REFERENCES organizations(id),
  agent_type            VARCHAR(30)  NOT NULL DEFAULT 'agency',
  -- VALUES: agency | travel_agent | uber | taxi | personal | online
  name                  VARCHAR(255) NOT NULL,
  contact_person        VARCHAR(255),
  phone                 VARCHAR(50),
  whatsapp              VARCHAR(50),
  email                 VARCHAR(255),
  commission_type       VARCHAR(30)  NOT NULL DEFAULT 'pct',
  -- VALUES: pct | fixed_per_booking | credit_accumulate
  commission_rate       DECIMAL(6,4) NOT NULL DEFAULT 0,
  commission_base       VARCHAR(30)  NOT NULL DEFAULT 'hostess_gross',
  payment_cycle         VARCHAR(30)  NOT NULL DEFAULT 'monthly',
  payment_method        VARCHAR(50),
  bank_name             VARCHAR(100),
  bank_account          VARCHAR(100),
  bank_country          CHAR(2),
  swift_code            VARCHAR(20),
  preferred_currency    CHAR(3)      DEFAULT 'MYR',
  credit_balance        DECIMAL(15,4) NOT NULL DEFAULT 0,
  notes                 TEXT,
  is_active             BOOLEAN      NOT NULL DEFAULT true,
  created_at            TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at            TIMESTAMPTZ
);

-- ============================================================
-- 8. STAFF
-- ============================================================
CREATE TABLE IF NOT EXISTS staff (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  branch_id         UUID         NOT NULL REFERENCES branches(id),
  employee_code     VARCHAR(50)  UNIQUE,
  full_name         VARCHAR(255) NOT NULL,
  legal_name        VARCHAR(255),
  nationality       VARCHAR(100),
  id_type           VARCHAR(20),
  id_number         VARCHAR(100),
  id_expiry         DATE,
  phone             VARCHAR(50),
  whatsapp          VARCHAR(50),
  email             VARCHAR(255),
  password_hash     TEXT,
  role              VARCHAR(30)  NOT NULL,
  -- VALUES: super_admin|admin|branch_manager|manager|hostess|driver|kitchen|hall|general
  employment_type   VARCHAR(30)  NOT NULL DEFAULT 'full_time',
  -- VALUES: full_time | part_time | contract | freelance
  hire_date         DATE,
  contract_start    DATE,
  contract_end      DATE,
  base_salary       DECIMAL(15,4),
  salary_currency   CHAR(3)      DEFAULT 'MYR',
  commission_config JSONB,
  incentive_config  JSONB,
  penalty_applies   BOOLEAN      NOT NULL DEFAULT false,
  agent_id          UUID         REFERENCES agents(id),
  profile_photo     TEXT,
  bank_name         VARCHAR(100),
  bank_account      VARCHAR(100),
  bank_country      CHAR(2),
  swift_code        VARCHAR(20),
  preferred_currency CHAR(3)     DEFAULT 'MYR',
  notes             TEXT,
  is_active         BOOLEAN      NOT NULL DEFAULT true,
  created_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at        TIMESTAMPTZ
);
CREATE INDEX IF NOT EXISTS idx_staff_branch ON staff(branch_id);
CREATE INDEX IF NOT EXISTS idx_staff_role   ON staff(role);

-- ============================================================
-- 9. STAFF SCHEDULES
-- ============================================================
CREATE TABLE IF NOT EXISTS staff_schedules (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  staff_id       UUID        NOT NULL REFERENCES staff(id),
  branch_id      UUID        NOT NULL REFERENCES branches(id),
  day_of_week    SMALLINT    NOT NULL CHECK (day_of_week BETWEEN 0 AND 6),
  shift_start    TIME        NOT NULL,
  shift_end      TIME        NOT NULL,
  is_overnight   BOOLEAN     NOT NULL DEFAULT false,
  effective_from DATE        NOT NULL,
  effective_to   DATE,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ============================================================
-- 10. ATTENDANCE
-- ============================================================
CREATE TABLE IF NOT EXISTS attendance (
  id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  staff_id         UUID        NOT NULL REFERENCES staff(id),
  branch_id        UUID        NOT NULL REFERENCES branches(id),
  work_date        DATE        NOT NULL,
  scheduled_start  TIMESTAMPTZ,
  scheduled_end    TIMESTAMPTZ,
  clock_in         TIMESTAMPTZ,
  clock_out        TIMESTAMPTZ,
  status           VARCHAR(30) NOT NULL DEFAULT 'present',
  -- VALUES: present|absent|late|early_leave|no_show|day_off|sick
  late_minutes     INT         NOT NULL DEFAULT 0,
  early_leave_min  INT         NOT NULL DEFAULT 0,
  penalty_amount   DECIMAL(15,4) NOT NULL DEFAULT 0,
  penalty_reason   TEXT,
  notes            TEXT,
  approved_by      UUID        REFERENCES staff(id),
  approved_at      TIMESTAMPTZ,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE(staff_id, work_date)
);

-- ============================================================
-- 11. CUSTOMERS
-- ============================================================
CREATE TABLE IF NOT EXISTS customers (
  id                 UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  org_id             UUID         NOT NULL REFERENCES organizations(id),
  customer_code      VARCHAR(50)  UNIQUE,
  full_name          VARCHAR(255),
  phone              VARCHAR(50),
  whatsapp           VARCHAR(50),
  email              VARCHAR(255),
  password_hash      TEXT,
  nationality        VARCHAR(100),
  language_pref      VARCHAR(10)  DEFAULT 'en',
  referral_code_used VARCHAR(50),
  referral_source    VARCHAR(30),
  referral_agent_id  UUID         REFERENCES agents(id),
  credit_balance     DECIMAL(15,4) NOT NULL DEFAULT 0,
  payment_type       VARCHAR(30)  NOT NULL DEFAULT 'standard',
  -- VALUES: standard | monthly_billing | credit_account
  credit_limit       DECIMAL(15,4) NOT NULL DEFAULT 0,
  credit_due_day     SMALLINT,
  vip_tier           VARCHAR(20)  DEFAULT 'standard',
  notes              TEXT,
  is_active          BOOLEAN      NOT NULL DEFAULT true,
  created_at         TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  deleted_at         TIMESTAMPTZ
);

-- ============================================================
-- 12. RESERVATIONS
-- ============================================================
CREATE TABLE IF NOT EXISTS reservations (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reservation_no    VARCHAR(50)  UNIQUE NOT NULL,
  branch_id         UUID         NOT NULL REFERENCES branches(id),
  customer_id       UUID         REFERENCES customers(id),
  customer_name     VARCHAR(255),
  customer_phone    VARCHAR(50),
  guest_count       SMALLINT     NOT NULL DEFAULT 1,
  reservation_date  DATE         NOT NULL,
  start_time        TIMESTAMPTZ  NOT NULL,
  end_time          TIMESTAMPTZ,
  duration_hours    DECIMAL(4,2),
  room_id           UUID         REFERENCES rooms(id),
  status            VARCHAR(30)  NOT NULL DEFAULT 'tentative',
  -- STATE MACHINE: tentative→confirmed→checked_in→(extended→checked_in)→checked_out
  --                any→cancelled | confirmed→no_show
  booking_channel   VARCHAR(30)  NOT NULL DEFAULT 'walk_in',
  -- VALUES: walk_in|customer_app|whatsapp|telegram|phone|agent
  referral_code     VARCHAR(50),
  agent_id          UUID         REFERENCES agents(id),
  is_outcall        BOOLEAN      NOT NULL DEFAULT false,
  special_requests  TEXT,
  internal_notes    TEXT,
  deposit_amount    DECIMAL(15,4) NOT NULL DEFAULT 0,
  deposit_paid      BOOLEAN      NOT NULL DEFAULT false,
  deposit_paid_at   TIMESTAMPTZ,
  deposit_method    VARCHAR(30),
  confirmed_at      TIMESTAMPTZ,
  checked_in_at     TIMESTAMPTZ,
  checked_out_at    TIMESTAMPTZ,
  cancelled_at      TIMESTAMPTZ,
  cancellation_reason TEXT,
  no_show_at        TIMESTAMPTZ,
  created_by        UUID         REFERENCES staff(id),
  created_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_reservations_branch_date ON reservations(branch_id, reservation_date);
CREATE INDEX IF NOT EXISTS idx_reservations_status      ON reservations(status);

CREATE TABLE IF NOT EXISTS reservation_hostesses (
  id                       UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reservation_id           UUID         NOT NULL REFERENCES reservations(id),
  hostess_id               UUID         NOT NULL REFERENCES staff(id),
  is_primary               BOOLEAN      NOT NULL DEFAULT false,
  status                   VARCHAR(30)  NOT NULL DEFAULT 'assigned',
  commission_rate_snapshot DECIMAL(6,4),
  session_fee              DECIMAL(15,4),
  notes                    TEXT,
  assigned_by              UUID         REFERENCES staff(id),
  assigned_at              TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS reservation_pickups (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reservation_id UUID         NOT NULL REFERENCES reservations(id),
  driver_id      UUID         NOT NULL REFERENCES staff(id),
  pickup_address TEXT         NOT NULL,
  return_address TEXT,
  pickup_time    TIMESTAMPTZ  NOT NULL,
  return_time    TIMESTAMPTZ,
  pickup_fee     DECIMAL(15,4) NOT NULL DEFAULT 0,
  status         VARCHAR(30)  NOT NULL DEFAULT 'scheduled',
  notes          TEXT,
  created_at     TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

-- ============================================================
-- 13. ORDERS
-- ============================================================
CREATE TABLE IF NOT EXISTS orders (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_no       VARCHAR(50)   UNIQUE NOT NULL,
  reservation_id UUID          REFERENCES reservations(id),
  branch_id      UUID          NOT NULL REFERENCES branches(id),
  customer_id    UUID          REFERENCES customers(id),
  order_type     VARCHAR(30)   NOT NULL DEFAULT 'reservation',
  subtotal       DECIMAL(15,4) NOT NULL DEFAULT 0,
  discount_amount DECIMAL(15,4) NOT NULL DEFAULT 0,
  sst_amount     DECIMAL(15,4) NOT NULL DEFAULT 0,
  service_charge DECIMAL(15,4) NOT NULL DEFAULT 0,
  total_amount   DECIMAL(15,4) NOT NULL DEFAULT 0,
  currency       CHAR(3)       NOT NULL DEFAULT 'MYR',
  payment_status VARCHAR(30)   NOT NULL DEFAULT 'pending',
  -- VALUES: pending|partial|paid|credit|refunded|voided
  payment_method VARCHAR(30),
  -- VALUES: cash|qr_touchngo|qr_grabpay|fpx|card|credit_account|bank_transfer
  payment_ref    VARCHAR(100),
  payment_notes  TEXT,
  invoice_pdf_url TEXT,
  notes          TEXT,
  created_by     UUID          REFERENCES staff(id),

  created_at     TIMESTAMPTZ   NOT NULL DEFAULT NOW(),
  updated_at     TIMESTAMPTZ   NOT NULL DEFAULT NOW()

CREATE INDEX IF NOT EXISTS idx_orders_reservation ON orders(reservation_id);

CREATE TABLE IF NOT EXISTS order_items (
  id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  description  TEXT          NOT NULL,
```

```javascript
// MIGRATION: This becomes Program.cs + startup configuration in .NET

const app = express();

// Health check
app.get('/health', async (req, res) => {
  try {
    await pool.query('SELECT 1');
    res.json({ status: 'ok', db: 'connected', timestamp: new Date().toISOString() });
  } catch (err) {
    res.status(500).json({ status: 'error', db: 'disconnected' });
  }
});

const PORT = process.env.PORT || 3000;

  await runMigrations();
  logger.info(`KL Project server running on port ${PORT}`);
});
```
