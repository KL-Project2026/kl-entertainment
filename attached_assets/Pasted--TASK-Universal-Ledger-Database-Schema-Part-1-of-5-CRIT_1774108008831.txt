CREATE TABLE ledger_accounts (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id           UUID NOT NULL REFERENCES tenants(id) ON DELETE RESTRICT,
  branch_id           UUID REFERENCES branches(id) ON DELETE RESTRICT,

  -- 계정 주체 타입
  account_type        VARCHAR(30) NOT NULL,
  -- ALLOWED VALUES:
  -- 'hostess'       커미션/급여
  -- 'agency'        수수료 정산
  -- 'investor'      수익 배분 (집계 전용)
  -- 'driver'        픽업 수당
  -- 'staff'         일반 직원 급여
  -- 'branch'        지점 내부 계정
  -- 'corporate'     법인 고객 선불
  -- 'guest_wallet'  VIP 선불 크레딧
  CONSTRAINT chk_account_type CHECK (account_type IN (
    'hostess','agency','investor','driver',
    'staff','branch','corporate','guest_wallet'
  )),

  entity_id           UUID NOT NULL,
  entity_type         VARCHAR(30) NOT NULL,

  currency            CHAR(3) NOT NULL DEFAULT 'MYR',
  is_active           BOOLEAN NOT NULL DEFAULT TRUE,

  -- 잔액 캐시 (성능용 — 원장 재계산으로 언제든 복구 가능)
  balance_cache       NUMERIC(12,2) NOT NULL DEFAULT 0.00,
  balance_updated_at  TIMESTAMPTZ,

  -- 계정 설정 (account_type별 메타데이터)
  settings            JSONB DEFAULT '{}',
  -- hostess: {"advance_limit": 500, "payout_frequency": "biweekly"}
  -- investor: {"profit_share_pct": 30, "dividend_frequency": "monthly"}
  -- agency:   {"invoice_frequency": "monthly", "invoice_day": 1}

  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_by          UUID REFERENCES users(id),

  -- 동일 엔티티에 동일 타입 계정 중복 방지
  UNIQUE(tenant_id, account_type, entity_id)
);

CREATE INDEX idx_ledger_accounts_tenant    ON ledger_accounts(tenant_id);
CREATE INDEX idx_ledger_accounts_entity    ON ledger_accounts(entity_id, entity_type);
CREATE INDEX idx_ledger_accounts_branch    ON ledger_accounts(branch_id);
CREATE INDEX idx_ledger_accounts_type      ON ledger_accounts(account_type);

### Step 2: ledger_entries
```sql
CREATE TABLE ledger_entries (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id           UUID NOT NULL REFERENCES tenants(id) ON DELETE RESTRICT,
  branch_id           UUID REFERENCES branches(id),
  account_id          UUID NOT NULL REFERENCES ledger_accounts(id) ON DELETE RESTRICT,

  -- 복식부기 핵심 필드
  entry_type          VARCHAR(50) NOT NULL,
  -- Hostess:  'commission_session'|'commission_drink'|'commission_package'
  --           'tip'|'bonus'|'advance'|'advance_repayment'
  --           'penalty'|'agency_deduction'|'payout'
  -- Agency:   'fee_accrued'|'invoice_issued'|'payment_received'|'credit_note'
  -- Investor: 'profit_allocation'|'dividend_paid'|'capital_contribution'
  -- Driver:   'pickup_fee'|'expense_claim'|'payout'
  -- Staff:    'base_salary'|'allowance'|'deduction'|'payout'
  -- Common:   'manual_adjustment'|'opening_balance'|'fx_adjustment'

  direction           CHAR(2) NOT NULL,
  CONSTRAINT chk_direction CHECK (direction IN ('DR', 'CR')),
  -- CR = 계정에 이익 (수령)
  -- DR = 계정에서 차감 (지급/공제)

  amount              NUMERIC(12,2) NOT NULL,
  CONSTRAINT chk_amount_positive CHECK (amount > 0),

  currency            CHAR(3) NOT NULL DEFAULT 'MYR',

  -- 멀티통화 FX 환산
  reporting_currency  CHAR(3),
  reporting_amount    NUMERIC(12,2),
  fx_rate             NUMERIC(10,6),

  -- 원천 추적 (어떤 운영 이벤트에서 발생했나)
  source_type         VARCHAR(40),
  -- 'room_session'|'folio_item'|'agency_invoice'
  -- 'payout_batch'|'investor_report'|'shift'|'manual'
  source_id           UUID,

  -- 상대 계정 (복식부기 균형)
  contra_account_id   UUID REFERENCES ledger_accounts(id),

  -- Gross → Net 분리 (Hostess/Agency 커미션용)
  gross_amount        NUMERIC(12,2),
  agency_share        NUMERIC(12,2) DEFAULT 0,
  hostess_net         NUMERIC(12,2),
  CONSTRAINT chk_gross_split CHECK (
    gross_amount IS NULL OR
    ROUND(COALESCE(agency_share,0) + COALESCE(hostess_net,0), 2) 
    = ROUND(gross_amount, 2)
  ),

  -- 회계 기간
  effective_date      DATE NOT NULL,
  accounting_period   CHAR(7),  -- 'YYYY-MM' 형식

  -- 문서 연결 (급여명세서/결의서)
  document_type       VARCHAR(30),
  -- 'payslip'|'commission_statement'|'resolution'|'invoice'|'receipt'
  document_id         UUID,

  -- 상태 관리
  status              VARCHAR(20) NOT NULL DEFAULT 'posted',
  CONSTRAINT chk_entry_status CHECK (status IN (
    'pending','posted','disputed','reversed','void'
  )),

  -- 취소/역분개 연결
  reversed_by_entry_id UUID REFERENCES ledger_entries(id),
  reversal_reason     TEXT,

  -- 감사 추적 (카라오케 분쟁 대비 필수)
  posted_by           UUID NOT NULL REFERENCES users(id),
  approved_by         UUID REFERENCES users(id),
  approved_at         TIMESTAMPTZ,
  posted_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  audit_note          TEXT,
  ip_address          INET,

  description         TEXT,
  reference_no        VARCHAR(100),

  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 성능 인덱스
CREATE INDEX idx_ledger_entries_account    ON ledger_entries(account_id);
CREATE INDEX idx_ledger_entries_tenant     ON ledger_entries(tenant_id);
CREATE INDEX idx_ledger_entries_branch     ON ledger_entries(branch_id);
CREATE INDEX idx_ledger_entries_period     ON ledger_entries(accounting_period);
CREATE INDEX idx_ledger_entries_effective  ON ledger_entries(effective_date);
CREATE INDEX idx_ledger_entries_source     ON ledger_entries(source_type, source_id);
CREATE INDEX idx_ledger_entries_status     ON ledger_entries(status);
CREATE INDEX idx_ledger_entries_type       ON ledger_entries(entry_type);
```

CREATE TABLE payslips (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id             UUID NOT NULL REFERENCES tenants(id),
  branch_id             UUID REFERENCES branches(id),
  account_id            UUID NOT NULL REFERENCES ledger_accounts(id),

  entity_type           VARCHAR(20) NOT NULL,
  CONSTRAINT chk_payslip_entity CHECK (
    entity_type IN ('hostess','driver','staff')
  ),
  entity_id             UUID NOT NULL,
  entity_name           VARCHAR(200),   -- 스냅샷 (이름 변경 대비)

  -- 급여 기간
  period_start          DATE NOT NULL,
  period_end            DATE NOT NULL,
  pay_date              DATE,

  -- 집계 금액 (ledger_entries에서 자동 계산)
  gross_earnings        NUMERIC(10,2) NOT NULL DEFAULT 0,
  total_deductions      NUMERIC(10,2) NOT NULL DEFAULT 0,
  net_payable           NUMERIC(10,2) NOT NULL DEFAULT 0,
  CONSTRAINT chk_net_payable CHECK (
    ROUND(net_payable, 2) = 
    ROUND(gross_earnings - total_deductions, 2)
  ),

  -- 항목 상세 스냅샷 (PDF 생성용)
  earnings_breakdown    JSONB NOT NULL DEFAULT '{}',
  /*
  {
    "commission_session":  1200.00,
    "commission_drink":     340.00,
    "commission_package":   150.00,
    "tip":                   80.00,
    "bonus":                200.00,
    "pickup_fee":           120.00
  }
  */
  deductions_breakdown  JSONB NOT NULL DEFAULT '{}',
  /*
  {
    "advance_repayment":   300.00,
    "agency_share":        350.00,
    "penalty":              50.00,
    "tax_withholding":      80.00
  }
  */

  -- 연결된 ledger_entry IDs
  ledger_entry_ids      UUID[] DEFAULT '{}',

  -- 문서
  pdf_url               TEXT,
  pdf_generated_at      TIMESTAMPTZ,

  -- 확인/서명
  acknowledged_at       TIMESTAMPTZ,
  acknowledged_by       UUID REFERENCES users(id),

  -- 상태
  status                VARCHAR(20) NOT NULL DEFAULT 'draft',
  CONSTRAINT chk_payslip_status CHECK (status IN (
    'draft','issued','acknowledged','disputed','paid','cancelled'

  dispute_note          TEXT,
  created_at            TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at            TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_by            UUID REFERENCES users(id),

  -- 동일 기간 중복 방지

### Step 5: 집계 뷰 생성
```sql
-- 계정별 실시간 잔액 뷰
CREATE OR REPLACE VIEW v_account_balances AS
SELECT
  la.id                 AS account_id,
  la.tenant_id,
  la.branch_id,
  la.account_type,
  la.entity_id,
  la.entity_type,
  la.currency,
  COALESCE(SUM(
    CASE WHEN le.direction = 'CR' THEN le.amount
         WHEN le.direction = 'DR' THEN -le.amount
    END
  ), 0.00)              AS current_balance,
  COUNT(le.id)          AS transaction_count,
  MAX(le.posted_at)     AS last_transaction_at,
  la.is_active
FROM ledger_accounts la
LEFT JOIN ledger_entries le
  ON le.account_id = la.id
  AND le.status = 'posted'
GROUP BY
  la.id, la.tenant_id, la.branch_id,
  la.account_type, la.entity_id,
  la.entity_type, la.currency, la.is_active;

-- 기간별 커미션 분배 현황 뷰 (Investor 보고 제외)
CREATE OR REPLACE VIEW v_commission_distribution AS
SELECT
  le.tenant_id,
  le.branch_id,
  le.effective_date,
  le.accounting_period,
  la.account_type,
  la.entity_id,
  le.entry_type,
  SUM(le.gross_amount)    AS total_gross,
  SUM(le.agency_share)    AS total_agency_share,
  SUM(le.hostess_net)     AS total_hostess_net,
  SUM(CASE WHEN le.direction='CR' THEN le.amount 
           ELSE -le.amount END) AS net_amount,
  COUNT(le.id)            AS entry_count
FROM ledger_entries le
JOIN ledger_accounts la ON la.id = le.account_id
WHERE le.status = 'posted'
GROUP BY
  le.tenant_id, le.branch_id,
  le.effective_date, le.accounting_period,
  la.account_type, la.entity_id, le.entry_type;

-- 미지급 잔액 뷰 (월별 정산 준비용)
CREATE OR REPLACE VIEW v_unpaid_balances AS
SELECT
  la.tenant_id,
  la.branch_id,
  la.account_type,
  la.entity_id,
  la.entity_type,
  la.currency,
  COALESCE(SUM(
    CASE WHEN le.direction = 'CR' THEN le.amount
         WHEN le.direction = 'DR' THEN -le.amount END
  ), 0.00)               AS unpaid_balance,
  MIN(le.effective_date) AS oldest_entry_date,
  COUNT(le.id)           AS pending_entries
FROM ledger_accounts la
LEFT JOIN ledger_entries le
  ON le.account_id = la.id
  AND le.status = 'posted'
  AND le.document_type IS NULL   -- 아직 급여명세서 미생성
GROUP BY
  la.tenant_id, la.branch_id,
  la.account_type, la.entity_id,
  la.entity_type, la.currency
HAVING COALESCE(SUM(
    CASE WHEN le.direction = 'CR' THEN le.amount
         WHEN le.direction = 'DR' THEN -le.amount END
  ), 0.00) > 0;
```
